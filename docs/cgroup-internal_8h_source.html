235static inline void put_css_set(struct css_set *cset)
236{
237 unsigned long flags;
238
239 /*
240 * Ensure that the refcount doesn't hit zero while any readers
241 * can see it. Similar to atomic_dec_and_lock(), but for an
242 * rwlock
243 */
244 if (refcount_dec_not_one(&cset->refcount))
245 return;
246
247 spin_lock_irqsave(&css_set_lock, flags);
248 put_css_set_locked(cset);
249 spin_unlock_irqrestore(&css_set_lock, flags);
250}
