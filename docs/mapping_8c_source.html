43static void dmam_release(struct device *dev, void *res)
44{
45 struct dma_devres *this = res;
46
47 dma_free_attrs(dev, this->size, this->vaddr, this->dma_handle,
48 this->attrs);
49}

54static int dmam_match(struct device *dev, void *res, void *match_data)
55{
56 struct dma_devres *this = res, *match = match_data;
57
58 if (this->vaddr == match->vaddr) {
59 WARN_ON(this->size != match->size ||
60 this->dma_handle != match->dma_handle);
61 return 1;
62 }
63 return 0;
64}

105void *dmam_alloc_attrs(struct device *dev, size_t size, dma_addr_t *dma_handle,
106 gfp_t gfp, unsigned long attrs)
107{
108 struct dma_devres *dr;
109 void *vaddr;
110
111 dr = devres_alloc(dmam_release, sizeof(*dr), gfp);
112 if (!dr)
113 return NULL;
114
115 vaddr = dma_alloc_attrs(dev, size, dma_handle, gfp, attrs);
116 if (!vaddr) {
117 devres_free(dr);
118 return NULL;
119 }
120
121 dr->vaddr = vaddr;
122 dr->dma_handle = *dma_handle;
123 dr->size = size;
124 dr->attrs = attrs;
125
126 devres_add(dev, dr);
127
128 return vaddr;
129}
130EXPORT_SYMBOL(dmam_alloc_attrs);

135static bool dma_go_direct(struct device *dev, dma_addr_t mask,
136 const struct dma_map_ops *ops)

179dma_addr_t dma_map_phys(struct device *dev, phys_addr_t phys, size_t size,
180 enum dma_data_direction dir, unsigned long attrs)
181{
182 const struct dma_map_ops *ops = get_dma_ops(dev);
183 bool is_mmio = attrs & DMA_ATTR_MMIO;
184 dma_addr_t addr = DMA_MAPPING_ERROR;
185
186 BUG_ON(!valid_dma_direction(dir));
187
188 if (WARN_ON_ONCE(!dev->dma_mask))
189 return DMA_MAPPING_ERROR;
190
191 if (!dev_is_dma_coherent(dev) && (attrs & DMA_ATTR_REQUIRE_COHERENT))
192 return DMA_MAPPING_ERROR;
193
194 if (dma_map_direct(dev, ops) ||
195 (!is_mmio && arch_dma_map_phys_direct(dev, phys + size)))
196 addr = dma_direct_map_phys(dev, phys, size, dir, attrs);
197 else if (use_dma_iommu(dev))
198 addr = iommu_dma_map_phys(dev, phys, size, dir, attrs);
199 else if (ops->map_phys)
200 addr = ops->map_phys(dev, phys, size, dir, attrs);
201
202 if (!is_mmio)
203 kmsan_handle_dma(phys, size, dir);
204 trace_dma_map_phys(dev, phys, addr, size, dir, attrs);
205 debug_dma_map_phys(dev, phys, size, dir, addr, attrs);
206
207 return addr;
208}
209EXPORT_SYMBOL_GPL(dma_map_phys);

269static int __dma_map_sg_attrs(struct device *dev, struct scatterlist *sg,
270 int nents, enum dma_data_direction dir, unsigned long attrs)
271{
272 const struct dma_map_ops *ops = get_dma_ops(dev);
273 int ents;
274
275 BUG_ON(!valid_dma_direction(dir));
276
277 if (!dev_is_dma_coherent(dev) && (attrs & DMA_ATTR_REQUIRE_COHERENT))
278 return -EOPNOTSUPP;
279
280 if (WARN_ON_ONCE(!dev->dma_mask))
281 return 0;
282
283 if (dma_map_direct(dev, ops) ||
284 arch_dma_map_sg_direct(dev, sg, nents))
285 ents = dma_direct_map_sg(dev, sg, nents, dir, attrs);
286 else if (use_dma_iommu(dev))
287 ents = iommu_dma_map_sg(dev, sg, nents, dir, attrs);
288 else
289 ents = ops->map_sg(dev, sg, nents, dir, attrs);
290
291 if (ents > 0) {
292 kmsan_handle_dma_sg(sg, nents, dir);
293 trace_dma_map_sg(dev, sg, nents, ents, dir, attrs);
294 debug_dma_map_sg(dev, sg, nents, ents, dir, attrs);
295 } else if (WARN_ON_ONCE(ents != -EINVAL && ents != -ENOMEM &&
296 ents != -EIO && ents != -EREMOTEIO)) {
297 trace_dma_map_sg_err(dev, sg, nents, ents, dir, attrs);
298 return -EIO;
299 }
300
301 return ents;
302}

506bool __dma_need_sync(struct device *dev, dma_addr_t dma_addr)
507{
508 const struct dma_map_ops *ops = get_dma_ops(dev);
509
510 if (dma_map_direct(dev, ops))
511 /*
512 * dma_skip_sync could've been reset on first SWIOTLB buffer
513 * mapping, but @dma_addr is not necessary an SWIOTLB buffer.
514 * In this case, fall back to more granular check.
515 */
516 return dma_direct_need_sync(dev, dma_addr);
517 return true;
518}
519EXPORT_SYMBOL_GPL(__dma_need_sync);

545static void dma_setup_need_sync(struct device *dev)
546{
547 const struct dma_map_ops *ops = get_dma_ops(dev);
548
549 if (dma_map_direct(dev, ops) || use_dma_iommu(dev))
550 /*
551 * dma_skip_sync will be reset to %false on first SWIOTLB buffer
552 * mapping, if any. During the device initialization, it's
553 * enough to check only for the DMA coherence.
554 */
555 dev->dma_skip_sync = dev_is_dma_coherent(dev);
556 else if (!ops->sync_single_for_device && !ops->sync_single_for_cpu &&
557 !ops->sync_sg_for_device && !ops->sync_sg_for_cpu)
558 /*
559 * Synchronization is not possible when none of DMA sync ops
560 * is set.
561 */
562 dev->dma_skip_sync = true;
563 else
564 dev->dma_skip_sync = false;
565}

709void *dma_alloc_attrs(struct device *dev, size_t size, dma_addr_t *dma_handle,
710 gfp_t flag, unsigned long attrs)
711{
712 const struct dma_map_ops *ops = get_dma_ops(dev);
713 void *cpu_addr;
714
715 WARN_ON_ONCE(!dev->coherent_dma_mask);
716
722 if (WARN_ON_ONCE(flag & __GFP_COMP))
723 return NULL;
724
725 if (dma_alloc_from_dev_coherent(dev, size, dma_handle, &cpu_addr)) {
726 trace_dma_alloc(dev, cpu_addr, *dma_handle, size,
727 DMA_BIDIRECTIONAL, flag, attrs);
728 return cpu_addr;
729 }
730
732 flag &= ~(__GFP_DMA | __GFP_DMA32 | __GFP_HIGHMEM);
733
734 if (dma_alloc_direct(dev, ops) || arch_dma_alloc_direct(dev)) {
735 cpu_addr = dma_direct_alloc(dev, size, dma_handle, flag, attrs);
736 } else if (use_dma_iommu(dev)) {
737 cpu_addr = iommu_dma_alloc(dev, size, dma_handle, flag, attrs);
738 } else if (ops->alloc) {
739 cpu_addr = ops->alloc(dev, size, dma_handle, flag, attrs);
740 } else {
741 trace_dma_alloc(dev, NULL, 0, size, DMA_BIDIRECTIONAL, flag,
742 attrs);
743 return NULL;
744 }
745
746 trace_dma_alloc(dev, cpu_addr, *dma_handle, size, DMA_BIDIRECTIONAL,
747 flag, attrs);
748 debug_dma_alloc_coherent(dev, size, *dma_handle, cpu_addr, attrs);
749 return cpu_addr;
750}
751EXPORT_SYMBOL(dma_alloc_attrs);

756void dma_free_attrs(struct device *dev, size_t size, void *cpu_addr,
757 dma_addr_t dma_handle, unsigned long attrs)
758{
759 const struct dma_map_ops *ops = get_dma_ops(dev);
760
761 if (dma_release_from_dev_coherent(dev, get_order(size), cpu_addr))
762 return;
770 WARN_ON(irqs_disabled());
771
772 trace_dma_free(dev, cpu_addr, dma_handle, size, DMA_BIDIRECTIONAL,
773 attrs);
774 if (!cpu_addr)
775 return;
776
777 debug_dma_free_coherent(dev, size, cpu_addr, dma_handle);
778 if (dma_alloc_direct(dev, ops) || arch_dma_free_direct(dev, dma_handle))
779 dma_direct_free(dev, size, cpu_addr, dma_handle, attrs);
780 else if (use_dma_iommu(dev))
781 iommu_dma_free(dev, size, cpu_addr, dma_handle, attrs);
782 else if (ops->free)
783 ops->free(dev, size, cpu_addr, dma_handle, attrs);
784}

790static struct page *__dma_alloc_pages(struct device *dev, size_t size,
791 dma_addr_t *dma_handle, enum dma_data_direction dir, gfp_t gfp)
792{
793 const struct dma_map_ops *ops = get_dma_ops(dev);
794
795 if (WARN_ON_ONCE(!dev->coherent_dma_mask))
796 return NULL;
797 if (WARN_ON_ONCE(gfp & (__GFP_DMA | __GFP_DMA32 | __GFP_HIGHMEM)))
798 return NULL;
799 if (WARN_ON_ONCE(gfp & __GFP_COMP))
800 return NULL;
801
802 size = PAGE_ALIGN(size);
803 if (dma_alloc_direct(dev, ops))
804 return dma_direct_alloc_pages(dev, size, dma_handle, dir, gfp);
805 if (use_dma_iommu(dev))
806 return dma_common_alloc_pages(dev, size, dma_handle, dir, gfp);
807 if (!ops->alloc_pages_op)
808 return NULL;
809 return ops->alloc_pages_op(dev, size, dma_handle, dir, gfp);
810}

863int dma_mmap_pages(struct device *dev, struct vm_area_struct *vma,
864 size_t size, struct page *page)
865{
866 unsigned long count = PAGE_ALIGN(size) >> PAGE_SHIFT;
867
868 if (vma->vm_pgoff >= count || vma_pages(vma) > count - vma->vm_pgoff)
869 return -ENXIO;
870 return remap_pfn_range(vma, vma->vm_start,
871 page_to_pfn(page) + vma->vm_pgoff,
872 vma_pages(vma) << PAGE_SHIFT, vma->vm_page_prot);
873}

879static struct sg_table *alloc_single_sgt(struct device *dev, size_t size,
880 enum dma_data_direction dir, gfp_t gfp)
881{
882 struct sg_table *sgt;
883 struct page *page;
884
885 sgt = kmalloc_obj(*sgt, gfp);
886 if (!sgt)
887 return NULL;
888 if (sg_alloc_table(sgt, 1, gfp))
889 goto out_free_sgt;
890 page = __dma_alloc_pages(dev, size, &sgt->sgl->dma_address, dir, gfp);
891 if (!page)
892 goto out_free_table;
893 sg_set_page(sgt->sgl, page, PAGE_ALIGN(size), 0);
894 sg_dma_len(sgt->sgl) = sgt->sgl->length;
895 return sgt;
896out_free_table:
897 sg_free_table(sgt);
898out_free_sgt:
899 kfree(sgt);
900 return NULL;
901}

906struct sg_table *dma_alloc_noncontiguous(struct device *dev, size_t size,
907 enum dma_data_direction dir, gfp_t gfp, unsigned long attrs)
908{
909 struct sg_table *sgt;
910
911 if (WARN_ON_ONCE(attrs & ~DMA_ATTR_ALLOC_SINGLE_PAGES))
912 return NULL;
913 if (WARN_ON_ONCE(gfp & __GFP_COMP))
914 return NULL;
915
916 if (use_dma_iommu(dev))
917 sgt = iommu_dma_alloc_noncontiguous(dev, size, dir, gfp, attrs);
918 else
919 sgt = alloc_single_sgt(dev, size, dir, gfp);
920
921 if (sgt) {
922 sgt->nents = 1;
923 trace_dma_alloc_sgt(dev, sgt, size, dir, gfp, attrs);
924 debug_dma_map_sg(dev, sgt->sgl, sgt->orig_nents, 1, dir, attrs);
925 } else {
926 trace_dma_alloc_sgt_err(dev, NULL, 0, size, dir, gfp, attrs);
927 }
928 return sgt;
929}

999static int dma_supported(struct device *dev, u64 mask)
1000{
1001 const struct dma_map_ops *ops = get_dma_ops(dev);
1002
1003 if (use_dma_iommu(dev)) {
1004 if (WARN_ON(ops))
1005 return false;
1006 return true;
1007 }
1008
1009 /*
1010 * ->dma_supported sets and clears the bypass flag, so ignore it here
1011 * and always call into the method if there is one.
1012 */
1013 if (ops) {
1014 if (!ops->dma_supported)
1015 return true;
1016 return ops->dma_supported(dev, mask);
1017 }
1018
1019 return dma_direct_supported(dev, mask);
1020}
