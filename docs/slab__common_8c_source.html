161static unsigned int calculate_alignment(slab_flags_t flags,
162 unsigned int align, unsigned int size)
163{
164 /*
165 * If the user wants hardware cache aligned objects then follow that
166 * suggestion if the object is sufficiently large.
167 *
168 * The hardware cache alignment cannot override the specified
169 * alignment though. If that is greater then use it.
170 */
171 if (flags & SLAB_HWCACHE_ALIGN) {
172 unsigned int ralign;
173
174 ralign = cache_line_size();
175 while (size <= ralign / 2)
176 ralign /= 2;
177 align = max(align, ralign);
178 }
179
180 align = max(align, arch_slab_minalign());
181
182 return ALIGN(align, sizeof(void *));
183}

191int slab_unmergeable(struct kmem_cache *s)
192{
193 if (slab_nomerge || (s->flags & SLAB_NEVER_MERGE))
194 return 1;
195
196 if (s->ctor)
197 return 1;
198
199#ifdef CONFIG_HARDENED_USERCOPY
200 if (s->usersize)
201 return 1;
202#endif
203
204 /*
205 * We may have set a slab to be unmergeable during bootstrap.
206 */
207 if (s->refcount < 0)
208 return 1;
209
210 return 0;
211}

236static struct kmem_cache *find_mergeable(unsigned int size, slab_flags_t flags,
237 const char *name, struct kmem_cache_args *args)
238{
239 struct kmem_cache *s;
240 unsigned int align;
241
242 flags = kmem_cache_flags(flags, name);
243 if (slab_args_unmergeable(args, flags))
244 return NULL;
245
246 size = ALIGN(size, sizeof(void *));
247 align = calculate_alignment(flags, args->align, size);
248 size = ALIGN(size, align);
249
250 list_for_each_entry_reverse(s, &slab_caches, list) {
251 if (slab_unmergeable(s))
252 continue;
253
254 if (size > s->size)
255 continue;
256
257 if ((flags & SLAB_MERGE_SAME) != (s->flags & SLAB_MERGE_SAME))
258 continue;
259 /*
260 * Check if alignment is compatible.
261 * Courtesy of Adrian Drzewiecki
262 */
263 if ((s->size & ~(align - 1)) != s->size)
264 continue;
265
266 if (s->size - size >= sizeof(void *))
267 continue;
268
269 return s;
270 }
271 return NULL;
272}

277static struct kmem_cache *create_cache(const char *name,
278 unsigned int object_size,
279 struct kmem_cache_args *args,
280 slab_flags_t flags)
281{
282 struct kmem_cache *s;
283 int err;
284
285 /* If a custom freelist pointer is requested make sure it's sane. */
286 err = -EINVAL;
287 if (args->use_freeptr_offset &&
288 (args->freeptr_offset >= object_size ||
289 (!(flags & SLAB_TYPESAFE_BY_RCU) && !args->ctor) ||
290 !IS_ALIGNED(args->freeptr_offset, __alignof__(freeptr_t))))
291 goto out;
292
293 err = -ENOMEM;
294 s = kmem_cache_zalloc(kmem_cache, GFP_KERNEL);
295 if (!s)
296 goto out;
297 err = do_kmem_cache_create(s, name, object_size, args, flags);
298 if (err)
299 goto out_free_cache;
300
301 s->refcount = 1;
302 list_add(&s->list, &slab_caches);
303 return s;
304
305out_free_cache:
306 kmem_cache_free(kmem_cache, s);
307out:
308 return ERR_PTR(err);
309}

314static struct kmem_cache *
315__kmem_cache_alias(const char *name, unsigned int size, slab_flags_t flags,
316 struct kmem_cache_args *args)
317{
318 struct kmem_cache *s;
319
320 s = find_mergeable(size, flags, name, args);
321 if (s) {
322 if (sysfs_slab_alias(s, name))
323 pr_err("SLUB: Unable to add cache alias %s to sysfs\n",
324 name);
325
326 s->refcount++;
327
328 /*
329 * Adjust the object sizes so that we clear
330 * the complete object on kzalloc.
331 */
332 s->object_size = max(s->object_size, size);
333 s->inuse = max(s->inuse, ALIGN(size, sizeof(void *)));
334 }
335
336 return s;
337}

369struct kmem_cache *__kmem_cache_create_args(const char *name,
370 unsigned int object_size,
371 struct kmem_cache_args *args,
372 slab_flags_t flags)
373{
374 struct kmem_cache *s = NULL;
375 const char *cache_name;
376 int err;
377
378#ifdef CONFIG_SLUB_DEBUG
379 /*
380 * If no slab_debug was enabled globally, the static key is not yet
381 * enabled by setup_slub_debug(). Enable it if the cache is being
382 * created with any of the debugging flags passed explicitly.
383 * It's also possible that this is the first cache created with
384 * SLAB_STORE_USER and we should init stack_depot for it.
385 */
386 if (flags & SLAB_DEBUG_FLAGS)
387 static_branch_enable(&slub_debug_enabled);
388 if (flags & SLAB_STORE_USER)
389 stack_depot_init();
390#else
391 flags &= ~SLAB_DEBUG_FLAGS;
392#endif
393
394 /*
395 * Caches with specific capacity are special enough. It's simpler to
396 * make them unmergeable.
397 */
398 if (args->sheaf_capacity)
399 flags |= SLAB_NO_MERGE;
400
401 mutex_lock(&slab_mutex);
402
403 err = kmem_cache_sanity_check(name, object_size);
404 if (err) {
405 goto out_unlock;
406 }
407
408 if (flags & ~SLAB_FLAGS_PERMITTED) {
409 err = -EINVAL;
410 goto out_unlock;
411 }
412
413 /* Fail closed on bad usersize of useroffset values. */
414 if (!IS_ENABLED(CONFIG_HARDENED_USERCOPY) ||
415 WARN_ON(!args->usersize && args->useroffset) ||
416 WARN_ON(object_size < args->usersize ||
417 object_size - args->usersize < args->useroffset))
418 args->usersize = args->useroffset = 0;
419
420 s = __kmem_cache_alias(name, object_size, flags, args);
421 if (s)
422 goto out_unlock;
423
424 cache_name = kstrdup_const(name, GFP_KERNEL);
425 if (!cache_name) {
426 err = -ENOMEM;
427 goto out_unlock;
428 }
429
430 args->align = calculate_alignment(flags, args->align, object_size);
431 s = create_cache(cache_name, object_size, args, flags);
432 if (IS_ERR(s)) {
433 err = PTR_ERR(s);
434 kfree_const(cache_name);
435 }
436
437out_unlock:
438 mutex_unlock(&slab_mutex);
439
440 if (err) {
441 if (flags & SLAB_PANIC)
442 panic("%s: Failed to create slab '%s'. Error %d\n",
443 __func__, name, err);
444 else {
445 pr_warn("%s(%s) failed with error %d\n",
446 __func__, name, err);
447 dump_stack();
448 }
449 return NULL;
450 }
451 return s;
452}
453EXPORT_SYMBOL(__kmem_cache_create_args);

479kmem_buckets *kmem_buckets_create(const char *name, slab_flags_t flags,
480 unsigned int useroffset,
481 unsigned int usersize,
482 void (*ctor)(void *))
483{
484 unsigned long mask = 0;
485 unsigned int idx;
486 kmem_buckets *b;
487
488 BUILD_BUG_ON(ARRAY_SIZE(kmalloc_caches[KMALLOC_NORMAL]) > BITS_PER_LONG);
489
490 /*
491 * When the separate buckets API is not built in, just return
492 * a non-NULL value for the kmem_buckets pointer, which will be
493 * unused when performing allocations.
494 */
495 if (!IS_ENABLED(CONFIG_SLAB_BUCKETS))
496 return ZERO_SIZE_PTR;
497
498 if (WARN_ON(!kmem_buckets_cache))
499 return NULL;
500
501 b = kmem_cache_alloc(kmem_buckets_cache, GFP_KERNEL|__GFP_ZERO);
502 if (WARN_ON(!b))
503 return NULL;
504
505 flags |= SLAB_NO_MERGE;
506
507 for (idx = 0; idx < ARRAY_SIZE(kmalloc_caches[KMALLOC_NORMAL]); idx++) {
508 char *short_size, *cache_name;
509 unsigned int cache_useroffset, cache_usersize;
510 unsigned int size, aligned_idx;
511
512 if (!kmalloc_caches[KMALLOC_NORMAL][idx])
513 continue;
514
515 size = kmalloc_caches[KMALLOC_NORMAL][idx]->object_size;
516 if (!size)
517 continue;
518
519 short_size = strchr(kmalloc_caches[KMALLOC_NORMAL][idx]->name, '-');
520 if (WARN_ON(!short_size))
521 goto fail;
522
523 if (useroffset >= size) {
524 cache_useroffset = 0;
525 cache_usersize = 0;
526 } else {
527 cache_useroffset = useroffset;
528 cache_usersize = min(size - cache_useroffset, usersize);
529 }
530
531 aligned_idx = __kmalloc_index(size, false);
532 if (!(*b)[aligned_idx]) {
533 cache_name = kasprintf(GFP_KERNEL, "%s-%s", name, short_size + 1);
534 if (WARN_ON(!cache_name))
535 goto fail;
536 (*b)[aligned_idx] = kmem_cache_create_usercopy(cache_name, size,
537 0, flags, cache_useroffset,
538 cache_usersize, ctor);
539 kfree(cache_name);
540 if (WARN_ON(!(*b)[aligned_idx]))
541 goto fail;
542 set_bit(aligned_idx, &mask);
543 }
544 if (idx != aligned_idx)
545 (*b)[idx] = (*b)[aligned_idx];
546 }
547
548 return b;
549
550fail:
551 for_each_set_bit(idx, &mask, ARRAY_SIZE(kmalloc_caches[KMALLOC_NORMAL]))
552 kmem_cache_destroy((*b)[idx]);
553 kmem_cache_free(kmem_buckets_cache, b);
554
555 return NULL;
556}

590void kmem_cache_destroy(struct kmem_cache *s)
591{
592 int err;
593
594 if (unlikely(!s) || !kasan_check_byte(s))
595 return;
596
597 /* in-flight kfree_rcu()'s may include objects from our cache */
598 kvfree_rcu_barrier_on_cache(s);
599
600 if (IS_ENABLED(CONFIG_SLUB_RCU_DEBUG) &&
601 (s->flags & SLAB_TYPESAFE_BY_RCU)) {
602 /*
603 * Under CONFIG_SLUB_RCU_DEBUG, when objects in a
604 * SLAB_TYPESAFE_BY_RCU slab are freed, SLUB will internally
605 * defer their freeing with call_rcu().
606 * Wait for such call_rcu() invocations here before actually
607 * destroying the cache.
608 *
609 * It doesn't matter that we haven't looked at the slab refcount
610 * yet - slabs with SLAB_TYPESAFE_BY_RCU can't be merged, so
611 * the refcount should be 1 here.
612 */
613 rcu_barrier();
614 }
615
616 /* Wait for deferred work from kmalloc/kfree_nolock() */
617 defer_free_barrier();
618
619 cpus_read_lock();
620 mutex_lock(&slab_mutex);
621
622 s->refcount--;
623 if (s->refcount) {
624 mutex_unlock(&slab_mutex);
625 cpus_read_unlock();
626 return;
627 }
628
629 /* free asan quarantined objects */
630 kasan_cache_shutdown(s);
631
632 err = __kmem_cache_shutdown(s);
633 if (!slab_in_kunit_test())
634 WARN(err, "%s %s: Slab cache still has objects when called from %pS",
635 __func__, s->name, (void *)_RET_IP_);
636
637 list_del(&s->list);
638
639 mutex_unlock(&slab_mutex);
640 cpus_read_unlock();
641
642 if (slab_state >= FULL)
643 sysfs_slab_unlink(s);
644 debugfs_slab_release(s);
645
646 if (err)
647 return;
648
649 if (s->flags & SLAB_TYPESAFE_BY_RCU)
650 rcu_barrier();
651
652 kmem_cache_release(s);
653}
654EXPORT_SYMBOL(kmem_cache_destroy);

713bool kmem_dump_obj(void *object)
714{
715 char *cp = IS_ENABLED(CONFIG_MMU) ? "" : "/vmalloc";
716 int i;
717 struct slab *slab;
718 unsigned long ptroffset;
719 struct kmem_obj_info kp = { };
720
721 /* Some arches consider ZERO_SIZE_PTR to be a valid address. */
722 if (object < (void *)PAGE_SIZE || !virt_addr_valid(object))
723 return false;
724 slab = virt_to_slab(object);
725 if (!slab)
726 return false;
727
728 kmem_obj_info(&kp, object, slab);
729 if (kp.kp_slab_cache)
730 pr_cont(" slab%s %s", cp, kp.kp_slab_cache->name);
731 else
732 pr_cont(" slab%s", cp);
733 if (is_kfence_address(object))
734 pr_cont(" (kfence)");
735 if (kp.kp_objp)
736 pr_cont(" start %px", kp.kp_objp);
737 if (kp.kp_data_offset)
738 pr_cont(" data offset %lu", kp.kp_data_offset);
739 if (kp.kp_objp) {
740 ptroffset = ((char *)object - (char *)kp.kp_objp) - kp.kp_data_offset;
741 pr_cont(" pointer offset %lu", ptroffset);
742 }
743 if (kp.kp_slab_cache && kp.kp_slab_cache->object_size)
744 pr_cont(" size %u", kp.kp_slab_cache->object_size);
745 if (kp.kp_ret)
746 pr_cont(" allocated at %pS\n", kp.kp_ret);
747 else
748 pr_cont("\n");
749 for (i = 0; i < ARRAY_SIZE(kp.kp_stack); i++) {
750 if (!kp.kp_stack[i])
751 break;
752 pr_info(" %pS\n", kp.kp_stack[i]);
753 }
754
755 if (kp.kp_free_stack[0])
756 pr_cont(" Free path:\n");
757
758 for (i = 0; i < ARRAY_SIZE(kp.kp_free_stack); i++) {
759 if (!kp.kp_free_stack[i])
760 break;
761 pr_info(" %pS\n", kp.kp_free_stack[i]);
762 }
763
764 return true;
765}

774void __init create_boot_cache(struct kmem_cache *s, const char *name,
775 unsigned int size, slab_flags_t flags,
776 unsigned int useroffset, unsigned int usersize)
777{
778 int err;
779 unsigned int align = ARCH_KMALLOC_MINALIGN;
780 struct kmem_cache_args kmem_args = {};
781
782 /*
783 * kmalloc caches guarantee alignment of at least the largest
784 * power-of-two divisor of the size. For power-of-two sizes,
785 * it is the size itself.
786 */
787 if (flags & SLAB_KMALLOC)
788 align = max(align, 1U << (ffs(size) - 1));
789 kmem_args.align = calculate_alignment(flags, align, size);
790
791#ifdef CONFIG_HARDENED_USERCOPY
792 kmem_args.useroffset = useroffset;
793 kmem_args.usersize = usersize;
794#endif
795
796 err = do_kmem_cache_create(s, name, size, &kmem_args, flags);
797
798 if (err)
799 panic("Creation of kmalloc slab %s size=%u failed. Reason %d\n",
800 name, size, err);
801
802 s->refcount = -1; /* Exempt from merging for now */
803}

871size_t kmalloc_size_roundup(size_t size)
872{
873 if (size && size <= KMALLOC_MAX_CACHE_SIZE) {
874 /*
875 * The flags don't matter since size_index is common to all.
876 * Neither does the caller for just getting ->object_size.
877 */
878 return kmalloc_slab(size, NULL, GFP_KERNEL, 0)->object_size;
879 }
880
881 /* Above the smaller buckets, size is a multiple of page size. */
882 if (size && size <= KMALLOC_MAX_SIZE)
883 return PAGE_SIZE << get_order(size);
884
885 /*
886 * Return 'size' for 0 - kmalloc() returns ZERO_SIZE_PTR
887 * and very large size - kmalloc() may fail.
888 */
889 return size;
890
891}

991void __init setup_kmalloc_cache_index_table(void)
992{
993 unsigned int i;
994
995 BUILD_BUG_ON(KMALLOC_MIN_SIZE > 256 ||
996 !is_power_of_2(KMALLOC_MIN_SIZE));
997
998 for (i = 8; i < KMALLOC_MIN_SIZE; i += 8) {
999 unsigned int elem = size_index_elem(i);
1000
1001 if (elem >= ARRAY_SIZE(kmalloc_size_index))
1002 break;
1003 kmalloc_size_index[elem] = KMALLOC_SHIFT_LOW;
1004 }
1005
1006 if (KMALLOC_MIN_SIZE >= 64) {
1007 /*
1008 * The 96 byte sized cache is not used if the alignment
1009 * is 64 byte.
1010 */
1011 for (i = 64 + 8; i <= 96; i += 8)
1012 kmalloc_size_index[size_index_elem(i)] = 7;
1013
1014 }
1015
1016 if (KMALLOC_MIN_SIZE >= 128) {
1017 /*
1018 * The 192 byte sized cache is not used if the alignment
1019 * is 128 byte. Redirect kmalloc to use the 256 byte cache
1020 * instead.
1021 */
1022 for (i = 128 + 8; i <= 192; i += 8)
1023 kmalloc_size_index[size_index_elem(i)] = 8;
1024 }
1025}

1044static void __init
1045new_kmalloc_cache(int idx, enum kmalloc_cache_type type)
1046{
1047 slab_flags_t flags = 0;
1048 unsigned int minalign = __kmalloc_minalign();
1049 unsigned int aligned_size = kmalloc_info[idx].size;
1050 int aligned_idx = idx;
1051
1052 if ((KMALLOC_RECLAIM != KMALLOC_NORMAL) && (type == KMALLOC_RECLAIM)) {
1053 flags |= SLAB_RECLAIM_ACCOUNT;
1054 } else if (IS_ENABLED(CONFIG_MEMCG) && (type == KMALLOC_CGROUP)) {
1055 if (mem_cgroup_kmem_disabled()) {
1056 kmalloc_caches[type][idx] = kmalloc_caches[KMALLOC_NORMAL][idx];
1057 return;
1058 }
1059 flags |= SLAB_ACCOUNT;
1060 } else if (IS_ENABLED(CONFIG_ZONE_DMA) && (type == KMALLOC_DMA)) {
1061 flags |= SLAB_CACHE_DMA;
1062 }
1063
1064#ifdef CONFIG_RANDOM_KMALLOC_CACHES
1065 if (type >= KMALLOC_RANDOM_START && type <= KMALLOC_RANDOM_END)
1066 flags |= SLAB_NO_MERGE;
1067#endif
1068
1069 /*
1070 * If CONFIG_MEMCG is enabled, disable cache merging for
1071 * KMALLOC_NORMAL caches.
1072 */
1073 if (IS_ENABLED(CONFIG_MEMCG) && (type == KMALLOC_NORMAL))
1074 flags |= SLAB_NO_MERGE;
1075
1076 if (minalign > ARCH_KMALLOC_MINALIGN) {
1077 aligned_size = ALIGN(aligned_size, minalign);
1078 aligned_idx = __kmalloc_index(aligned_size, false);
1079 }
1080
1081 if (!kmalloc_caches[type][aligned_idx])
1082 kmalloc_caches[type][aligned_idx] = create_kmalloc_cache(
1083 kmalloc_info[aligned_idx].name[type],
1084 aligned_size, flags);
1085 if (idx != aligned_idx)
1086 kmalloc_caches[type][idx] = kmalloc_caches[type][aligned_idx];
1087}

1098{

1101
1102 /*
1103 * Including KMALLOC_CGROUP if CONFIG_MEMCG defined
1104 */

1743static void kfree_rcu_work(struct work_struct *work)
1744{
1745 unsigned long flags;
1746 struct kvfree_rcu_bulk_data *bnode, *n;
1747 struct list_head bulk_head[FREE_N_CHANNELS];
1748 struct rcu_head *head;
1749 struct kfree_rcu_cpu *krcp;
1750 struct kfree_rcu_cpu_work *krwp;
1751 struct rcu_gp_oldstate head_gp_snap;
1752 int i;
1753
1754 krwp = container_of(to_rcu_work(work),
1755 struct kfree_rcu_cpu_work, rcu_work);
1756 krcp = krwp->krcp;
1757
1758 raw_spin_lock_irqsave(&krcp->lock, flags);
1759 // Channels 1 and 2.
1760 for (i = 0; i < FREE_N_CHANNELS; i++)
1761 list_replace_init(&krwp->bulk_head_free[i], &bulk_head[i]);
1762
1763 // Channel 3.
1764 head = krwp->head_free;
1765 krwp->head_free = NULL;
1766 head_gp_snap = krwp->head_free_gp_snap;
1767 raw_spin_unlock_irqrestore(&krcp->lock, flags);
1768
1769 // Handle the first two channels.
1770 for (i = 0; i < FREE_N_CHANNELS; i++) {
1771 // Start from the tail page, so a GP is likely passed for it.
1772 list_for_each_entry_safe(bnode, n, &bulk_head[i], list)
1773 kvfree_rcu_bulk(krcp, bnode, i);
1774 }
1775
1776 /*
1777 * This is used when the "bulk" path can not be used for the
1778 * double-argument of kvfree_rcu(). This happens when the
1779 * page-cache is empty, which means that objects are instead
1780 * queued on a linked list through their rcu_head structures.
1781 * This list is named "Channel 3".
1782 */
1783 if (head && !WARN_ON_ONCE(!poll_state_synchronize_rcu_full(&head_gp_snap)))
1784 kvfree_rcu_list(head);
1785}

1887static void
1888kvfree_rcu_drain_ready(struct kfree_rcu_cpu *krcp)
1889{
1890 struct list_head bulk_ready[FREE_N_CHANNELS];
1891 struct kvfree_rcu_bulk_data *bnode, *n;
1892 struct rcu_head *head_ready = NULL;
1893 unsigned long flags;
1894 int i;
1895
1896 raw_spin_lock_irqsave(&krcp->lock, flags);
1897 for (i = 0; i < FREE_N_CHANNELS; i++) {
1898 INIT_LIST_HEAD(&bulk_ready[i]);
1899
1900 list_for_each_entry_safe_reverse(bnode, n, &krcp->bulk_head[i], list) {
1901 if (!poll_state_synchronize_rcu_full(&bnode->gp_snap))
1902 break;
1903
1904 atomic_sub(bnode->nr_records, &krcp->bulk_count[i]);
1905 list_move(&bnode->list, &bulk_ready[i]);
1906 }
1907 }
1908
1909 if (krcp->head && poll_state_synchronize_rcu(krcp->head_gp_snap)) {
1910 head_ready = krcp->head;
1911 atomic_set(&krcp->head_count, 0);
1912 WRITE_ONCE(krcp->head, NULL);
1913 }
1914 raw_spin_unlock_irqrestore(&krcp->lock, flags);
1915
1916 for (i = 0; i < FREE_N_CHANNELS; i++) {
1917 list_for_each_entry_safe(bnode, n, &bulk_ready[i], list)
1918 kvfree_rcu_bulk(krcp, bnode, i);
1919 }
1920
1921 if (head_ready)
1922 kvfree_rcu_list(head_ready);
1923}

1931static bool
1932kvfree_rcu_queue_batch(struct kfree_rcu_cpu *krcp)
1933{
1934 unsigned long flags;
1935 bool queued = false;
1936 int i, j;
1937
1938 raw_spin_lock_irqsave(&krcp->lock, flags);
1939
1940 // Attempt to start a new batch.
1941 for (i = 0; i < KFREE_N_BATCHES; i++) {
1942 struct kfree_rcu_cpu_work *krwp = &(krcp->krw_arr[i]);
1943
1944 // Try to detach bulk_head or head and attach it, only when
1945 // all channels are free. Any channel is not free means at krwp
1946 // there is on-going rcu work to handle krwp's free business.
1947 if (need_wait_for_krwp_work(krwp))
1948 continue;
1949
1950 // kvfree_rcu_drain_ready() might handle this krcp, if so give up.
1951 if (need_offload_krc(krcp)) {
1952 // Channel 1 corresponds to the SLAB-pointer bulk path.
1953 // Channel 2 corresponds to vmalloc-pointer bulk path.
1954 for (j = 0; j < FREE_N_CHANNELS; j++) {
1955 if (list_empty(&krwp->bulk_head_free[j])) {
1956 atomic_set(&krcp->bulk_count[j], 0);
1957 list_replace_init(&krcp->bulk_head[j],
1958 &krwp->bulk_head_free[j]);
1959 }
1960 }
1961
1962 // Channel 3 corresponds to both SLAB and vmalloc
1963 // objects queued on the linked list.
1964 if (!krwp->head_free) {
1965 krwp->head_free = krcp->head;
1966 get_state_synchronize_rcu_full(&krwp->head_free_gp_snap);
1967 atomic_set(&krcp->head_count, 0);
1968 WRITE_ONCE(krcp->head, NULL);
1969 }
1970
1971 // One work is per one batch, so there are three
1972 // "free channels", the batch can handle. Break
1973 // the loop since it is done with this CPU thus
1974 // queuing an RCU work is _always_ success here.
1975 queued = queue_rcu_work(rcu_reclaim_wq, &krwp->rcu_work);
1976 WARN_ON_ONCE(!queued);
1977 break;
1978 }
1979 }
1980
1981 raw_spin_unlock_irqrestore(&krcp->lock, flags);
1982 return queued;
1983}

1991static void kfree_rcu_monitor(struct work_struct *work)
1992{
1993 struct kfree_rcu_cpu *krcp = container_of(work,
1994 struct kfree_rcu_cpu, monitor_work.work);
1995
1996 // Drain ready for reclaim.
1997 kvfree_rcu_drain_ready(krcp);
1998
1999 // Queue a batch for a rest.
2000 kvfree_rcu_queue_batch(krcp);
2001
2002 // If there is nothing to detach, it means that our job is
2003 // successfully done here. In case of having at least one
2004 // of the channels that is still busy we should rearm the
2005 // work to repeat an attempt. Because previous batches are
2006 // still in progress.
2007 if (need_offload_krc(krcp))
2008 schedule_delayed_monitor_work(krcp);
2009}

2014static void fill_page_cache_func(struct work_struct *work)
2015{
2016 struct kvfree_rcu_bulk_data *bnode;
2017 struct kfree_rcu_cpu *krcp =
2018 container_of(work, struct kfree_rcu_cpu,
2019 page_cache_work.work);
2020 unsigned long flags;
2021 int nr_pages;
2022 bool pushed;
2023 int i;
2024
2025 nr_pages = atomic_read(&krcp->backoff_page_cache_fill) ?
2026 1 : rcu_min_cached_objs;
2027
2028 for (i = READ_ONCE(krcp->nr_bkv_objs); i < nr_pages; i++) {
2029 bnode = (struct kvfree_rcu_bulk_data *)
2030 __get_free_page(GFP_KERNEL | __GFP_NORETRY | __GFP_NOMEMALLOC | __GFP_NOWARN);
2031
2032 if (!bnode)
2033 break;
2034
2035 raw_spin_lock_irqsave(&krcp->lock, flags);
2036 pushed = put_cached_bnode(krcp, bnode);
2037 raw_spin_unlock_irqrestore(&krcp->lock, flags);
2038
2039 if (!pushed) {
2040 free_page((unsigned long) bnode);
2041 break;
2042 }
2043 }
2044
2045 atomic_set(&krcp->work_in_progress, 0);
2046 atomic_set(&krcp->backoff_page_cache_fill, 0);
2047}

2060static inline bool
2061add_ptr_to_bulk_krc_lock(struct kfree_rcu_cpu **krcp,
2062 unsigned long *flags, void *ptr, bool can_alloc)
2063{
2064 struct kvfree_rcu_bulk_data *bnode;
2065 int idx;
2066
2067 *krcp = krc_this_cpu_lock(flags);
2068 if (unlikely(!(*krcp)->initialized))
2069 return false;
2070
2071 idx = !!is_vmalloc_addr(ptr);
2072 bnode = list_first_entry_or_null(&(*krcp)->bulk_head[idx],
2073 struct kvfree_rcu_bulk_data, list);
2074
2075 /* Check if a new block is required. */
2076 if (!bnode || bnode->nr_records == KVFREE_BULK_MAX_ENTR) {
2077 bnode = get_cached_bnode(*krcp);
2078 if (!bnode && can_alloc) {
2079 krc_this_cpu_unlock(*krcp, *flags);
2080
2081 // __GFP_NORETRY - allows a light-weight direct reclaim
2082 // what is OK from minimizing of fallback hitting point of
2083 // view. Apart of that it forbids any OOM invoking what is
2084 // also beneficial since we are about to release memory soon.
2085 //
2086 // __GFP_NOMEMALLOC - prevents from consuming of all the
2087 // memory reserves. Please note we have a fallback path.
2088 //
2089 // __GFP_NOWARN - it is supposed that an allocation can
2090 // be failed under low memory or high memory pressure
2091 // scenarios.
2092 bnode = (struct kvfree_rcu_bulk_data *)
2093 __get_free_page(GFP_KERNEL | __GFP_NORETRY | __GFP_NOMEMALLOC | __GFP_NOWARN);
2094 raw_spin_lock_irqsave(&(*krcp)->lock, *flags);
2095 }
2096
2097 if (!bnode)
2098 return false;
2099
2100 // Initialize the new block and attach it.
2101 bnode->nr_records = 0;
2102 list_add(&bnode->list, &(*krcp)->bulk_head[idx]);
2103 }
2104
2105 // Finally insert and update the GP for this page.
2106 bnode->nr_records++;
2107 bnode->records[bnode->nr_records - 1] = ptr;
2108 get_state_synchronize_rcu_full(&bnode->gp_snap);
2109 atomic_inc(&(*krcp)->bulk_count[idx]);
2110
2111 return true;
2112}

2181void kvfree_call_rcu(struct rcu_head *head, void *ptr)
2182{
2183 unsigned long flags;
2184 struct kfree_rcu_cpu *krcp;
2185 bool success;
2186
2187 /*
2188 * Please note there is a limitation for the head-less
2189 * variant, that is why there is a clear rule for such
2190 * objects: it can be used from might_sleep() context
2191 * only. For other places please embed an rcu_head to
2192 * your data.
2193 */
2194 if (!head)
2195 might_sleep();
2196
2197 if (!IS_ENABLED(CONFIG_PREEMPT_RT) && kfree_rcu_sheaf(ptr))
2198 return;
2199
2200 // Queue the object but don't yet schedule the batch.
2201 if (debug_rcu_head_queue(ptr)) {
2202 // Probable double kfree_rcu(), just leak.
2203 WARN_ONCE(1, "%s(): Double-freed call. rcu_head %p\n",
2204 __func__, head);
2205
2206 // Mark as success and leave.
2207 return;
2208 }
2209
2210 kasan_record_aux_stack(ptr);
2211 success = add_ptr_to_bulk_krc_lock(&krcp, &flags, ptr, !head);
2212 if (!success) {
2213 run_page_cache_worker(krcp);
2214
2215 if (head == NULL)
2216 // Inline if kvfree_rcu(one_arg) call.
2217 goto unlock_return;
2218
2219 head->func = ptr;
2220 head->next = krcp->head;
2221 WRITE_ONCE(krcp->head, head);
2222 atomic_inc(&krcp->head_count);
2223
2224 // Take a snapshot for this krcp.
2225 krcp->head_gp_snap = get_state_synchronize_rcu();
2226 success = true;
2227 }
2228
2229 /*
2230 * The kvfree_rcu() caller considers the pointer freed at this point
2231 * and likely removes any references to it. Since the actual slab
2232 * freeing (and kmemleak_free()) is deferred, tell kmemleak to ignore
2233 * this object (no scanning or false positives reporting).
2234 */
2235 kmemleak_ignore(ptr);
2236
2237 // Set timer to drain after KFREE_DRAIN_JIFFIES.
2238 if (rcu_scheduler_active == RCU_SCHEDULER_RUNNING)
2239 __schedule_delayed_monitor_work(krcp);
2240
2241unlock_return:
2242 krc_this_cpu_unlock(krcp, flags);
2243
2244 /*
2245 * Inline kvfree() after synchronize_rcu(). We can do
2246 * it from might_sleep() context only, so the current
2247 * CPU can pass the QS state.
2248 */
2249 if (!success) {
2250 debug_rcu_head_unqueue((struct rcu_head *) ptr);
2251 synchronize_rcu();
2252 kvfree(ptr);
2253 }
2254}
2255EXPORT_SYMBOL_GPL(kvfree_call_rcu);

2260static inline void __kvfree_rcu_barrier(void)
2261{
2262 struct kfree_rcu_cpu_work *krwp;
2263 struct kfree_rcu_cpu *krcp;
2264 bool queued;
2265 int i, cpu;
2266
2267 /*
2268 * Firstly we detach objects and queue them over an RCU-batch
2269 * for all CPUs. Finally queued works are flushed for each CPU.
2270 *
2271 * Please note. If there are outstanding batches for a particular
2272 * CPU, those have to be finished first following by queuing a new.
2273 */
2274 for_each_possible_cpu(cpu) {
2275 krcp = per_cpu_ptr(&krc, cpu);
2276
2277 /*
2278 * Check if this CPU has any objects which have been queued for a
2279 * new GP completion. If not(means nothing to detach), we are done
2280 * with it. If any batch is pending/running for this "krcp", below
2281 * per-cpu flush_rcu_work() waits its completion(see last step).
2282 */
2283 if (!need_offload_krc(krcp))
2284 continue;
2285
2286 while (1) {
2287 /*
2288 * If we are not able to queue a new RCU work it means:
2289 * - batches for this CPU are still in flight which should
2290 * be flushed first and then repeat;
2291 * - no objects to detach, because of concurrency.
2292 */
2293 queued = kvfree_rcu_queue_batch(krcp);
2294
2295 /*
2296 * Bail out, if there is no need to offload this "krcp"
2297 * anymore. As noted earlier it can run concurrently.
2298 */
2299 if (queued || !need_offload_krc(krcp))
2300 break;
2301
2302 /* There are ongoing batches. */
2303 for (i = 0; i < KFREE_N_BATCHES; i++) {
2304 krwp = &(krcp->krw_arr[i]);
2305 flush_rcu_work(&krwp->rcu_work);
2306 }
2307 }
2308 }
2309
2310 /*
2311 * Now we guarantee that all objects are flushed.
2312 */
2313 for_each_possible_cpu(cpu) {
2314 krcp = per_cpu_ptr(&krc, cpu);
2315
2316 /*
2317 * A monitor work can drain ready to reclaim objects
2318 * directly. Wait its completion if running or pending.
2319 */
2320 cancel_delayed_work_sync(&krcp->monitor_work);
2321
2322 for (i = 0; i < KFREE_N_BATCHES; i++) {
2323 krwp = &(krcp->krw_arr[i]);
2324 flush_rcu_work(&krwp->rcu_work);
2325 }
2326 }
2327}
