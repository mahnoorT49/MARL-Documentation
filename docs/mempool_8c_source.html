48static void poison_error(struct mempool *pool, void *element, size_t size,
49 size_t byte)
50{
51 const int nr = pool->curr_nr;
52 const int start = max_t(int, byte - (BITS_PER_LONG / 8), 0);
53 const int end = min_t(int, byte + (BITS_PER_LONG / 8), size);
54 int i;
55
56 pr_err("BUG: mempool element poison mismatch\n");
57 pr_err("Mempool %p size %zu\n", pool, size);
58 pr_err(" nr=%d @ %p: %s0x", nr, element, start > 0 ? "... " : "");
59 for (i = start; i < end; i++)
60 pr_cont("%x ", *(u8 *)(element + i));
61 pr_cont("%s\n", end < size ? "..." : "");
62 dump_stack();
63}

68static void __check_element(struct mempool *pool, void *element, size_t size)
69{
70 u8 *obj = element;
71 size_t i;
72
73 for (i = 0; i < size; i++) {
74 u8 exp = (i < size - 1) ? POISON_FREE : POISON_END;
75
76 if (obj[i] != exp) {
77 poison_error(pool, element, size, i);
78 return;
79 }
80 }
81 memset(obj, POISON_INUSE, size);
82}

87static void check_element(struct mempool *pool, void *element)
88{
89 /* Skip checking: KASAN might save its metadata in the element. */
90 if (kasan_enabled())
91 return;
92
93 /* Mempools backed by slab allocator */
94 if (pool->free == mempool_kfree) {
95 __check_element(pool, element, (size_t)pool->pool_data);
96 } else if (pool->free == mempool_free_slab) {
97 __check_element(pool, element, kmem_cache_size(pool->pool_data));
98 } else if (pool->free == mempool_free_pages) {
99 /* Mempools backed by page allocator */
100 int order = (int)(long)pool->pool_data;
101
102#ifdef CONFIG_HIGHMEM
103 for (int i = 0; i < (1 << order); i++) {
104 struct page *page = (struct page *)element;
105 void *addr = kmap_local_page(page + i);
106
107 __check_element(pool, addr, PAGE_SIZE);
108 kunmap_local(addr);
109 }
110#else
111 void *addr = page_address((struct page *)element);
112
113 __check_element(pool, addr, PAGE_SIZE << order);
114#endif
115 }
116}

121static void __poison_element(void *element, size_t size)
122{
123 u8 *obj = element;
124
125 memset(obj, POISON_FREE, size - 1);
126 obj[size - 1] = POISON_END;
127}

132static void poison_element(struct mempool *pool, void *element)
133{
134 /* Skip poisoning: KASAN might save its metadata in the element. */
135 if (kasan_enabled())
136 return;
137
138 /* Mempools backed by slab allocator */
139 if (pool->alloc == mempool_kmalloc) {
140 __poison_element(element, (size_t)pool->pool_data);
141 } else if (pool->alloc == mempool_alloc_slab) {
142 __poison_element(element, kmem_cache_size(pool->pool_data));
143 } else if (pool->alloc == mempool_alloc_pages) {
144 /* Mempools backed by page allocator */
145 int order = (int)(long)pool->pool_data;
146
147#ifdef CONFIG_HIGHMEM
148 for (int i = 0; i < (1 << order); i++) {
149 struct page *page = (struct page *)element;
150 void *addr = kmap_local_page(page + i);
151
152 __poison_element(addr, PAGE_SIZE);
153 kunmap_local(addr);
154 }
155#else
156 void *addr = page_address((struct page *)element);
157
158 __poison_element(addr, PAGE_SIZE << order);
159#endif
160 }
161}

174static __always_inline bool kasan_poison_element(struct mempool *pool,
175 void *element)
176{
177 if (pool->alloc == mempool_alloc_slab || pool->alloc == mempool_kmalloc)
178 return kasan_mempool_poison_object(element);
179 else if (pool->alloc == mempool_alloc_pages)
180 return kasan_mempool_poison_pages(element,
181 (unsigned long)pool->pool_data);
182 return true;
183}
184
188static void kasan_unpoison_element(struct mempool *pool, void *element)
189{
190 if (pool->alloc == mempool_kmalloc)
191 kasan_mempool_unpoison_object(element, (size_t)pool->pool_data);
192 else if (pool->alloc == mempool_alloc_slab)
193 kasan_mempool_unpoison_object(element,
194 kmem_cache_size(pool->pool_data));
195 else if (pool->alloc == mempool_alloc_pages)
196 kasan_mempool_unpoison_pages(element,
197 (unsigned long)pool->pool_data);
198}

203static __always_inline void add_element(struct mempool *pool, void *element)
204{
205 BUG_ON(pool->min_nr != 0 && pool->curr_nr >= pool->min_nr);
206 poison_element(pool, element);
207 if (kasan_poison_element(pool, element))
208 pool->elements[pool->curr_nr++] = element;
209}
210
214static void *remove_element(struct mempool *pool)
215{
216 void *element = pool->elements[--pool->curr_nr];
217
218 BUG_ON(pool->curr_nr < 0);
219 kasan_unpoison_element(pool, element);
220 check_element(pool, element);
221 return element;
222}

238void mempool_exit(struct mempool *pool)
239{
240 while (pool->curr_nr) {
241 void *element = remove_element(pool);
242 pool->free(element, pool->pool_data);
243 }
244 kfree(pool->elements);
245 pool->elements = NULL;
246}
247EXPORT_SYMBOL(mempool_exit);

274int mempool_init_node(struct mempool *pool, int min_nr,
275 mempool_alloc_t *alloc_fn, mempool_free_t *free_fn,
276 void *pool_data, gfp_t gfp_mask, int node_id)
277{
278 spin_lock_init(&pool->lock);
279 pool->min_nr = min_nr;
280 pool->pool_data = pool_data;
281 pool->alloc = alloc_fn;
282 pool->free = free_fn;
283 init_waitqueue_head(&pool->wait);
284 /*
285 * max() used here to ensure storage for at least 1 element to support
286 * zero minimum pool
287 */
288 pool->elements = kmalloc_array_node(max(1, min_nr), sizeof(void *),
289 gfp_mask, node_id);
290 if (!pool->elements)
291 return -ENOMEM;
292
293 /*
294 * First pre-allocate the guaranteed number of buffers,
295 * also pre-allocate 1 element for zero minimum pool.
296 */
297 while (pool->curr_nr < max(1, pool->min_nr)) {
298 void *element;
299
300 element = pool->alloc(gfp_mask, pool->pool_data);
301 if (unlikely(!element)) {
302 mempool_exit(pool);
303 return -ENOMEM;
304 }
305 add_element(pool, element);
306 }
307
308 return 0;
309}
310EXPORT_SYMBOL(mempool_init_node);

401int mempool_resize(struct mempool *pool, int new_min_nr)
402{
403 void *element;
404 void **new_elements;
405 unsigned long flags;
406
407 BUG_ON(new_min_nr <= 0);
408 might_sleep();
409
410 spin_lock_irqsave(&pool->lock, flags);
411 if (new_min_nr <= pool->min_nr) {
412 while (new_min_nr < pool->curr_nr) {
413 element = remove_element(pool);
414 spin_unlock_irqrestore(&pool->lock, flags);
415 pool->free(element, pool->pool_data);
416 spin_lock_irqsave(&pool->lock, flags);
417 }
418 pool->min_nr = new_min_nr;
419 goto out_unlock;
420 }
421 spin_unlock_irqrestore(&pool->lock, flags);
422
423 /* Grow the pool */
424 new_elements = kmalloc_objs(*new_elements, new_min_nr);
425 if (!new_elements)
426 return -ENOMEM;
427
428 spin_lock_irqsave(&pool->lock, flags);
429 if (unlikely(new_min_nr <= pool->min_nr)) {
430 /* Raced, other resize will do our work */
431 spin_unlock_irqrestore(&pool->lock, flags);
432 kfree(new_elements);
433 goto out;
434 }
435 memcpy(new_elements, pool->elements,
436 pool->curr_nr * sizeof(*new_elements));
437 kfree(pool->elements);
438 pool->elements = new_elements;
439 pool->min_nr = new_min_nr;
440
441 while (pool->curr_nr < pool->min_nr) {
442 spin_unlock_irqrestore(&pool->lock, flags);
443 element = pool->alloc(GFP_KERNEL, pool->pool_data);
444 if (!element)
445 goto out;
446 spin_lock_irqsave(&pool->lock, flags);
447 if (pool->curr_nr < pool->min_nr) {
448 add_element(pool, element);
449 } else {
450 spin_unlock_irqrestore(&pool->lock, flags);
451 pool->free(element, pool->pool_data); /* Raced */
452 goto out;
453 }
454 }
455out_unlock:
456 spin_unlock_irqrestore(&pool->lock, flags);
457out:
458 return 0;
459}
460EXPORT_SYMBOL(mempool_resize);

465static unsigned int mempool_alloc_from_pool(struct mempool *pool, void **elems,
466 unsigned int count, unsigned int allocated,
467 gfp_t gfp_mask)
468{
469 unsigned long flags;
470 unsigned int i;
471
472 spin_lock_irqsave(&pool->lock, flags);
473 if (unlikely(pool->curr_nr < count - allocated))
474 goto fail;
475 for (i = 0; i < count; i++) {
476 if (!elems[i]) {
477 elems[i] = remove_element(pool);
478 allocated++;
479 }
480 }
481 spin_unlock_irqrestore(&pool->lock, flags);
482
483 /* Paired with rmb in mempool_free(), read comment there. */
484 smp_wmb();
485
486 /*
487 * Update the allocation stack trace as this is more useful for
488 * debugging.
489 */
490 for (i = 0; i < count; i++)
491 kmemleak_update_trace(elems[i]);
492 return allocated;
493
494fail:
495 if (gfp_mask & __GFP_DIRECT_RECLAIM) {
496 DEFINE_WAIT(wait);
497
498 prepare_to_wait(&pool->wait, &wait, TASK_UNINTERRUPTIBLE);
499 spin_unlock_irqrestore(&pool->lock, flags);
500
501 /*
502 * Wait for someone else to return an element to @pool, but wake
503 * up occasionally as memory pressure might have reduced even
504 * and the normal allocation in alloc_fn could succeed even if
505 * no element was returned.
506 */
507 io_schedule_timeout(5 * HZ);
508 finish_wait(&pool->wait, &wait);
509 } else {
510 /* We must not sleep if __GFP_DIRECT_RECLAIM is not set. */
511 spin_unlock_irqrestore(&pool->lock, flags);
512 }
513
514 return allocated;
515}

527static inline gfp_t mempool_adjust_gfp(gfp_t *gfp_mask)
528{
529 *gfp_mask |= __GFP_NOMEMALLOC | __GFP_NORETRY | __GFP_NOWARN;
530 return *gfp_mask & ~(__GFP_DIRECT_RECLAIM | __GFP_IO);
531}

551int mempool_alloc_bulk_noprof(struct mempool *pool, void **elems,
552 unsigned int count, unsigned int allocated)
553{
554 gfp_t gfp_mask = GFP_KERNEL;
555 gfp_t gfp_temp = mempool_adjust_gfp(&gfp_mask);
556 unsigned int i = 0;
557
558 VM_WARN_ON_ONCE(count > pool->min_nr);
559 might_alloc(gfp_mask);
560
561 /*
562 * If an error is injected, fail all elements in a bulk allocation so
563 * that we stress the multiple elements missing path.
564 */
565 if (should_fail_ex(&fail_mempool_alloc_bulk, 1, FAULT_NOWARN)) {
566 pr_info("forcing mempool usage for %pS\n",
567 (void *)_RET_IP_);
568 goto use_pool;
569 }
570
571repeat_alloc:
572 /*
573 * Try to allocate the elements using the allocation callback first as
574 * that might succeed even when the caller's bulk allocation did not.
575 */
576 for (i = 0; i < count; i++) {
577 if (elems[i])
578 continue;
579 elems[i] = pool->alloc(gfp_temp, pool->pool_data);
580 if (unlikely(!elems[i]))
581 goto use_pool;
582 allocated++;
583 }
584
585 return 0;
586
587use_pool:
588 allocated = mempool_alloc_from_pool(pool, elems, count, allocated,
589 gfp_temp);
590 gfp_temp = gfp_mask;
591 goto repeat_alloc;
592}
593EXPORT_SYMBOL_GPL(mempool_alloc_bulk_noprof);

614void *mempool_alloc_noprof(struct mempool *pool, gfp_t gfp_mask)
615{
616 gfp_t gfp_temp = mempool_adjust_gfp(&gfp_mask);
617 void *element;
618
619 VM_WARN_ON_ONCE(gfp_mask & __GFP_ZERO);
620 might_alloc(gfp_mask);
621
622repeat_alloc:
623 if (should_fail_ex(&fail_mempool_alloc, 1, FAULT_NOWARN)) {
624 pr_info("forcing mempool usage for %pS\n",
625 (void *)_RET_IP_);
626 element = NULL;
627 } else {
628 element = pool->alloc(gfp_temp, pool->pool_data);
629 }
630
631 if (unlikely(!element)) {
632 /*
633 * Try to allocate an element from the pool.
634 *
635 * The first pass won't have __GFP_DIRECT_RECLAIM and won't
636 * sleep in mempool_alloc_from_pool. Retry the allocation
637 * with all flags set in that case.
638 */
639 if (!mempool_alloc_from_pool(pool, &element, 1, 0, gfp_temp)) {
640 if (gfp_temp != gfp_mask) {
641 gfp_temp = gfp_mask;
642 goto repeat_alloc;
643 }
644 if (gfp_mask & __GFP_DIRECT_RECLAIM) {
645 goto repeat_alloc;
646 }
647 }
648 }
649
650 return element;
651}
652EXPORT_SYMBOL(mempool_alloc_noprof);

696unsigned int mempool_free_bulk(struct mempool *pool, void **elems,
697 unsigned int count)
698{
699 unsigned long flags;
700 unsigned int freed = 0;
701 bool added = false;
702
703 /*
704 * Paired with the wmb in mempool_alloc(). The preceding read is
705 * for @element and the following @pool->curr_nr. This ensures
706 * that the visible value of @pool->curr_nr is from after the
707 * allocation of @element. This is necessary for fringe cases
708 * where @element was passed to this task without going through
709 * barriers.
710 *
711 * For example, assume @p is %NULL at the beginning and one task
712 * performs "p = mempool_alloc(...);" while another task is doing
713 * "while (!p) cpu_relax(); mempool_free(p, ...);". This function
714 * may end up using curr_nr value which is from before allocation
715 * of @p without the following rmb.
716 */
717 smp_rmb();
718
719 /*
720 * For correctness, we need a test which is guaranteed to trigger
721 * if curr_nr + #allocated == min_nr. Testing curr_nr < min_nr
722 * without locking achieves that and refilling as soon as possible
723 * is desirable.
724 *
725 * Because curr_nr visible here is always a value after the
726 * allocation of @element, any task which decremented curr_nr below
727 * min_nr is guaranteed to see curr_nr < min_nr unless curr_nr gets
728 * incremented to min_nr afterwards. If curr_nr gets incremented
729 * to min_nr after the allocation of @element, the elements
730 * allocated after that are subject to the same guarantee.
731 *
732 * Waiters happen iff curr_nr is 0 and the above guarantee also
733 * ensures that there will be frees which return elements to the
734 * pool waking up the waiters.
735 *
736 * For zero-minimum pools, curr_nr < min_nr (0 < 0) never succeeds,
737 * so waiters sleeping on pool->wait would never be woken by the
738 * wake-up path of previous test. This explicit check ensures the
739 * allocation of element when both min_nr and curr_nr are 0, and
740 * any active waiters are properly awakened.
741 */
742 if (unlikely(READ_ONCE(pool->curr_nr) < pool->min_nr)) {
743 spin_lock_irqsave(&pool->lock, flags);
744 while (pool->curr_nr < pool->min_nr && freed < count) {
745 add_element(pool, elems[freed++]);
746 added = true;
747 }
748 spin_unlock_irqrestore(&pool->lock, flags);
749 } else if (unlikely(pool->min_nr == 0 &&
750 READ_ONCE(pool->curr_nr) == 0)) {
751 /* Handle the min_nr = 0 edge case: */
752 spin_lock_irqsave(&pool->lock, flags);
753 if (likely(pool->curr_nr == 0)) {
754 add_element(pool, elems[freed++]);
755 added = true;
756 }
757 spin_unlock_irqrestore(&pool->lock, flags);
758 }
759
760 if (unlikely(added) && wq_has_sleeper(&pool->wait))
761 wake_up(&pool->wait);
762
763 return freed;
764}
765EXPORT_SYMBOL_GPL(mempool_free_bulk);
