79int cgroup_attach_task_all(struct task_struct *from, struct task_struct *tsk)
80{
81 struct cgroup_root *root;
82 int retval = 0;
83
84 cgroup_lock();
85 cgroup_attach_lock(CGRP_ATTACH_LOCK_GLOBAL, NULL);
86 for_each_root(root) {
87 struct cgroup *from_cgrp;
88
89 spin_lock_irq(&css_set_lock);
90 from_cgrp = task_cgroup_from_root(from, root);
91 spin_unlock_irq(&css_set_lock);
92
93 retval = cgroup_attach_task(from_cgrp, tsk, false);
94 if (retval)
95 break;
96 }
97 cgroup_attach_unlock(CGRP_ATTACH_LOCK_GLOBAL, NULL);
98 cgroup_unlock();
99
100 return retval;
101}

120int cgroup_transfer_tasks(struct cgroup *to, struct cgroup *from)
121{
122 DEFINE_CGROUP_MGCTX(mgctx);
123 struct cgrp_cset_link *link;
124 struct css_task_iter it;
125 struct task_struct *task;
126 int ret;
127
128 if (cgroup_on_dfl(to))
129 return -EINVAL;
130
131 ret = cgroup_migrate_vet_dst(to);
132 if (ret)
133 return ret;
134
135 cgroup_lock();
136
137 cgroup_attach_lock(CGRP_ATTACH_LOCK_GLOBAL, NULL);
138
139 /* all tasks in @from are being moved, all csets are source */
140 spin_lock_irq(&css_set_lock);
141 list_for_each_entry(link, &from->cset_links, cset_link)
142 cgroup_migrate_add_src(link->cset, to, &mgctx);
143 spin_unlock_irq(&css_set_lock);
144
145 ret = cgroup_migrate_prepare_dst(&mgctx);
146 if (ret)
147 goto out_err;
148
149 /*
150 * Migrate tasks one-by-one until @from is empty. This fails iff
151 * ->can_attach() fails.
152 */
153 do {
154 css_task_iter_start(&from->self, 0, &it);
155
156 do {
157 task = css_task_iter_next(&it);
158 } while (task && (task->flags & PF_EXITING));
159
160 if (task)
161 get_task_struct(task);
162 css_task_iter_end(&it);
163
164 if (task) {
165 ret = cgroup_migrate(task, false, &mgctx);
166 if (!ret)
167 TRACE_CGROUP_PATH(transfer_tasks, to, task, false);
168 put_task_struct(task);
169 }
170 } while (task && !ret);
171out_err:
172 cgroup_migrate_finish(&mgctx);
173 cgroup_attach_unlock(CGRP_ATTACH_LOCK_GLOBAL, NULL);
174 cgroup_unlock();
175 return ret;
176}

244static void cgroup_pidlist_destroy_work_fn(struct work_struct *work)
245{
246 struct delayed_work *dwork = to_delayed_work(work);
247 struct cgroup_pidlist *l = container_of(dwork, struct cgroup_pidlist,
248 destroy_dwork);
249 struct cgroup_pidlist *tofree = NULL;
250
251 mutex_lock(&l->owner->pidlist_mutex);
252
253 /*
254 * Destroy iff we didn't get queued again. The state won't change
255 * as destroy_dwork can only be queued while locked.
256 */
257 if (!delayed_work_pending(dwork)) {
258 list_del(&l->links);
259 kvfree(l->list);
260 put_pid_ns(l->key.ns);
261 tofree = l;
262 }
263
264 mutex_unlock(&l->owner->pidlist_mutex);
265 kfree(tofree);
266}

275static int pidlist_uniq(pid_t *list, int length)
276{
277 int src, dest = 1;
278
279 /*
280 * we presume the 0th element is unique, so i starts at 1. trivial
281 * edge cases first; no work needs to be done for either
282 */
283 if (length == 0 || length == 1)
284 return length;
285 /* src and dest walk down the list; dest counts unique elements */
286 for (src = 1; src < length; src++) {
287 /* find next unique element */
288 while (list[src] == list[src-1]) {
289 src++;
290 if (src == length)
291 goto after;
292 }
293 /* dest always points to where the next unique element goes */
294 list[dest] = list[src];
295 dest++;
296 }
297after:
298 return dest;
299}

321static struct cgroup_pidlist *cgroup_pidlist_find(struct cgroup *cgrp,
322 enum cgroup_filetype type)
323{
324 struct cgroup_pidlist *l;
325 /* don't need task_nsproxy() if we're looking at ourself */
326 struct pid_namespace *ns = task_active_pid_ns(current);
327
328 lockdep_assert_held(&cgrp->pidlist_mutex);
329
330 list_for_each_entry(l, &cgrp->pidlists, links)
331 if (l->key.type == type && l->key.ns == ns)
332 return l;
333 return NULL;
334}

345static struct cgroup_pidlist *cgroup_pidlist_find_create(struct cgroup *cgrp,
346 enum cgroup_filetype type)
347{
348 struct cgroup_pidlist *l;
349
350 lockdep_assert_held(&cgrp->pidlist_mutex);
351
352 l = cgroup_pidlist_find(cgrp, type);
353 if (l)
354 return l;
355
356 /* entry not found; create a new one */
357 l = kzalloc_obj(struct cgroup_pidlist);
358 if (!l)
359 return l;
360
361 INIT_DELAYED_WORK(&l->destroy_dwork, cgroup_pidlist_destroy_work_fn);
362 l->key.type = type;
363 /* don't need task_nsproxy() if we're looking at ourself */
364 l->key.ns = get_pid_ns(task_active_pid_ns(current));
365 l->owner = cgrp;
366 list_add(&l->links, &cgrp->pidlists);
367 return l;
368}

376static int pidlist_array_load(struct cgroup *cgrp, enum cgroup_filetype type,
377 struct cgroup_pidlist **lp)
378{
379 pid_t *array;
380 int length;
381 int pid, n = 0; /* used for populating the array */
382 struct css_task_iter it;
383 struct task_struct *tsk;
384 struct cgroup_pidlist *l;
385
386 lockdep_assert_held(&cgrp->pidlist_mutex);
387
388 /*
389 * If cgroup gets more users after we read count, we won't have
390 * enough space - tough. This race is indistinguishable to the
391 * caller from the case that the additional cgroup users didn't
392 * show up until sometime later on.
393 */
394 length = cgroup_task_count(cgrp);
395 array = kvmalloc_objs(pid_t, length);
396 if (!array)
397 return -ENOMEM;
398 /* now, populate the array */
399 css_task_iter_start(&cgrp->self, 0, &it);
400 while ((tsk = css_task_iter_next(&it))) {
401 if (unlikely(n == length))
402 break;
403 /* get tgid or pid for procs or tasks file respectively */
404 if (type == CGROUP_FILE_PROCS)
405 pid = task_tgid_vnr(tsk);
406 else
407 pid = task_pid_vnr(tsk);
408 if (pid > 0) /* make sure to only use valid results */
409 array[n++] = pid;
410 }
411 css_task_iter_end(&it);
412 length = n;
413 /* now sort & strip out duplicates (tgids or recycled thread PIDs) */
414 sort(array, length, sizeof(pid_t), cmppid, NULL);
415 length = pidlist_uniq(array, length);
416
417 l = cgroup_pidlist_find_create(cgrp, type);
418 if (!l) {
419 kvfree(array);
420 return -ENOMEM;
421 }
422
423 /* store array, freeing old if necessary */
424 kvfree(l->list);
425 l->list = array;
426 l->length = length;
427 *lp = l;
428 return 0;
429}

439static void *cgroup_pidlist_start(struct seq_file *s, loff_t *pos)
440{
441 /*
442 * Initially we receive a position value that corresponds to
443 * one more than the last pid shown (or 0 on the first call or
444 * after a seek to the start). Use a binary-search to find the
445 * next pid to display, if any
446 */
447 struct kernfs_open_file *of = s->private;
448 struct cgroup_file_ctx *ctx = of->priv;
449 struct cgroup *cgrp = seq_css(s)->cgroup;
450 struct cgroup_pidlist *l;
451 enum cgroup_filetype type = seq_cft(s)->private;
452 int index = 0, pid = *pos;
453 int *iter, ret;
454
455 mutex_lock(&cgrp->pidlist_mutex);
456
457 /*
458 * !NULL @ctx->procs1.pidlist indicates that this isn't the first
459 * start() after open. If the matching pidlist is around, we can use
460 * that. Look for it. Note that @ctx->procs1.pidlist can't be used
461 * directly. It could already have been destroyed.
462 */
463 if (ctx->procs1.pidlist)
464 ctx->procs1.pidlist = cgroup_pidlist_find(cgrp, type);
465
466 /*
467 * Either this is the first start() after open or the matching
468 * pidlist has been destroyed inbetween. Create a new one.
469 */
470 if (!ctx->procs1.pidlist) {
471 ret = pidlist_array_load(cgrp, type, &ctx->procs1.pidlist);
472 if (ret)
473 return ERR_PTR(ret);
474 }
475 l = ctx->procs1.pidlist;
476
477 if (pid) {
478 int end = l->length;
479
480 while (index < end) {
481 int mid = (index + end) / 2;
482 if (l->list[mid] == pid) {
483 index = mid;
484 break;
485 } else if (l->list[mid] < pid)
486 index = mid + 1;
487 else
488 end = mid;
489 }
490 }
491 /* If we're off the end of the array, we're done */
492 if (index >= l->length)
493 return NULL;
494 /* Update the abstract position to be the actual pid that we found */
495 iter = l->list + index;
496 *pos = *iter;
497 return iter;
498}

518static void *cgroup_pidlist_next(struct seq_file *s, void *v, loff_t *pos)
519{
520 struct kernfs_open_file *of = s->private;
521 struct cgroup_file_ctx *ctx = of->priv;
522 struct cgroup_pidlist *l = ctx->procs1.pidlist;
523 pid_t *p = v;
524 pid_t *end = l->list + l->length;
525 /*
526 * Advance to the next pid in the array. If this goes off the
527 * end, we're done
528 */
529 p++;
530 if (p >= end) {
531 (*pos)++;
532 return NULL;
533 } else {
534 *pos = *p;
535 return p;
536 }
537}

552static ssize_t __cgroup1_procs_write(struct kernfs_open_file *of,
553 char *buf, size_t nbytes, loff_t off,
554 bool threadgroup)
555{
556 struct cgroup *cgrp;
557 struct task_struct *task;
558 const struct cred *cred, *tcred;
559 ssize_t ret;
560 enum cgroup_attach_lock_mode lock_mode;
561
562 cgrp = cgroup_kn_lock_live(of->kn, false);
563 if (!cgrp)
564 return -ENODEV;
565
566 task = cgroup_procs_write_start(buf, threadgroup, &lock_mode);
567 ret = PTR_ERR_OR_ZERO(task);
568 if (ret)
569 goto out_unlock;
570
571 /*
572 * Even if we're attaching all tasks in the thread group, we only need
573 * to check permissions on one of them. Check permissions using the
574 * credentials from file open to protect against inherited fd attacks.
575 */
576 cred = of->file->f_cred;
577 tcred = get_task_cred(task);
578 if (!uid_eq(cred->euid, GLOBAL_ROOT_UID) &&
579 !uid_eq(cred->euid, tcred->uid) &&
580 !uid_eq(cred->euid, tcred->suid))
581 ret = -EACCES;
582 put_cred(tcred);
583 if (ret)
584 goto out_finish;
585
586 ret = cgroup_attach_task(cgrp, task, threadgroup);
587
588out_finish:
589 cgroup_procs_write_finish(task, lock_mode);
590out_unlock:
591 cgroup_kn_unlock(of->kn);
592
593 return ret ?: nbytes;
594}

812int cgroupstats_build(struct cgroupstats *stats, struct dentry *dentry)
813{
814 struct kernfs_node *kn = kernfs_node_from_dentry(dentry);
815 struct cgroup *cgrp;
816 struct css_task_iter it;
817 struct task_struct *tsk;
818
819 /* it should be kernfs_node belonging to cgroupfs and is a directory */
820 if (dentry->d_sb->s_type != &cgroup_fs_type || !kn ||
821 kernfs_type(kn) != KERNFS_DIR)
822 return -EINVAL;
823
824 /*
825 * We aren't being called from kernfs and there's no guarantee on
826 * @kn->priv's validity. For this and css_tryget_online_from_dir(),
827 * @kn->priv is RCU safe. Let's do the RCU dancing.
828 */
829 rcu_read_lock();
830 cgrp = rcu_dereference(*(void __rcu __force **)&kn->priv);
831 if (!cgrp || !cgroup_tryget(cgrp)) {
832 rcu_read_unlock();
833 return -ENOENT;
834 }
835 rcu_read_unlock();
836
837 css_task_iter_start(&cgrp->self, 0, &it);
838 while ((tsk = css_task_iter_next(&it))) {
839 switch (READ_ONCE(tsk->__state)) {
840 case TASK_RUNNING:
841 stats->nr_running++;
842 break;
843 case TASK_INTERRUPTIBLE:
844 stats->nr_sleeping++;
845 break;
846 case TASK_UNINTERRUPTIBLE:
847 stats->nr_uninterruptible++;
848 break;
849 case TASK_STOPPED:
850 stats->nr_stopped++;
851 break;
852 default:
853 if (tsk->in_iowait)
854 stats->nr_io_wait++;
855 break;
856 }
857 }
858 css_task_iter_end(&it);
859
860 cgroup_put(cgrp);
861 return 0;
862}

900void cgroup1_release_agent(struct work_struct *work)
901{
902 struct cgroup *cgrp =
903 container_of(work, struct cgroup, release_agent_work);
904 char *pathbuf, *agentbuf;
905 char *argv[3], *envp[3];
906 int ret;
907
908 /* snoop agent path and exit early if empty */
909 if (!cgrp->root->release_agent_path[0])
910 return;
911
912 /* prepare argument buffers */
913 pathbuf = kmalloc(PATH_MAX, GFP_KERNEL);
914 agentbuf = kmalloc(PATH_MAX, GFP_KERNEL);
915 if (!pathbuf || !agentbuf)
916 goto out_free;
917
918 spin_lock(&release_agent_path_lock);
919 strscpy(agentbuf, cgrp->root->release_agent_path, PATH_MAX);
920 spin_unlock(&release_agent_path_lock);
921 if (!agentbuf[0])
922 goto out_free;
923
924 ret = cgroup_path_ns(cgrp, pathbuf, PATH_MAX, &init_cgroup_ns);
925 if (ret < 0)
926 goto out_free;
927
928 argv[0] = agentbuf;
929 argv[1] = pathbuf;
930 argv[2] = NULL;
931
932 /* minimal command environment */
933 envp[0] = "HOME=/";
934 envp[1] = "PATH=/sbin:/bin:/usr/sbin:/usr/bin";
935 envp[2] = NULL;

1055int cgroup1_parse_param(struct fs_context *fc, struct fs_parameter *param)
1056{
1057 struct cgroup_fs_context *ctx = cgroup_fc2context(fc);
1058 struct cgroup_subsys *ss;
1059 struct fs_parse_result result;
1060 int opt, i;
1061
1062 opt = fs_parse(fc, cgroup1_fs_parameters, param, &result);
1063 if (opt == -ENOPARAM) {
1064 int ret;
1065
1066 ret = vfs_parse_fs_param_source(fc, param);
1067 if (ret != -ENOPARAM)
1068 return ret;
1069 for_each_subsys(ss, i) {
1070 if (strcmp(param->key, ss->legacy_name) ||
1071 cgroup1_subsys_absent(ss))
1072 continue;
1073 if (!cgroup_ssid_enabled(i) || cgroup1_ssid_disabled(i))
1074 return invalfc(fc, "Disabled controller '%s'",
1075 param->key);
1076 ctx->subsys_mask |= (1 << i);
1077 return 0;
1078 }
1079 return invalfc(fc, "Unknown subsys name '%s'", param->key);
1080 }
1081 if (opt < 0)
1082 return opt;
1083
1084 switch (opt) {
1085 case Opt_none:
1086 /* Explicitly have no subsystems */
1087 ctx->none = true;
1088 break;
1089 case Opt_all:
1090 ctx->all_ss = true;
1091 break;
1092 case Opt_noprefix:
1093 ctx->flags |= CGRP_ROOT_NOPREFIX;
1094 break;
1095 case Opt_clone_children:
1096 ctx->cpuset_clone_children = true;
1097 break;
1098 case Opt_cpuset_v2_mode:
1099 ctx->flags |= CGRP_ROOT_CPUSET_V2_MODE;
1100 break;
1101 case Opt_xattr:
1102 ctx->flags |= CGRP_ROOT_XATTR;
1103 break;
1104 case Opt_favordynmods:
1105 ctx->flags |= CGRP_ROOT_FAVOR_DYNMODS;
1106 break;
1107 case Opt_nofavordynmods:
1108 ctx->flags &= ~CGRP_ROOT_FAVOR_DYNMODS;
1109 break;
1110 case Opt_release_agent:
1111 /* Specifying two release agents is forbidden */
1112 if (ctx->release_agent)
1113 return invalfc(fc, "release_agent respecified");
1114 /*
1115 * Release agent gets called with all capabilities,
1116 * require capabilities to set release agent.
1117 */
1118 if ((fc->user_ns != &init_user_ns) || !capable(CAP_SYS_ADMIN))
1119 return invalfc(fc, "Setting release_agent not allowed");
1120 ctx->release_agent = param->string;
1121 param->string = NULL;
1122 break;
1123 case Opt_name:
1124 /* blocked by boot param? */
1125 if (cgroup_no_v1_named)
1126 return -ENOENT;
1127 /* Can't specify an empty name */
1128 if (!param->size)
1129 return invalfc(fc, "Empty name");
1130 if (param->size > MAX_CGROUP_ROOT_NAMELEN - 1)
1131 return invalfc(fc, "Name too long");
1132 /* Must match [\w.-]+ */
1133 for (i = 0; i < param->size; i++) {
1134 char c = param->string[i];
1135 if (isalnum(c))
1136 continue;
1137 if ((c == '.') || (c == '-') || (c == '_'))
1138 continue;
1139 return invalfc(fc, "Invalid name");
1140 }
1141 /* Specifying two names is forbidden */
1142 if (ctx->name)
1143 return invalfc(fc, "name respecified");
1144 ctx->name = param->string;
1145 param->string = NULL;
1146 break;
1147 }
1148 return 0;
1149}

1154static int check_cgroupfs_options(struct fs_context *fc)
1155{
1156 struct cgroup_fs_context *ctx = cgroup_fc2context(fc);
1157 u32 mask = U32_MAX;
1158 u32 enabled = 0;
1159 struct cgroup_subsys *ss;
1160 int i;
1161
1162#ifdef CONFIG_CPUSETS
1163 mask = ~((u32)1 << cpuset_cgrp_id);
1164#endif
1165 for_each_subsys(ss, i)
1166 if (cgroup_ssid_enabled(i) && !cgroup1_ssid_disabled(i) &&
1167 !cgroup1_subsys_absent(ss))
1168 enabled |= 1 << i;
1169
1170 ctx->subsys_mask &= enabled;
1171
1172 /*
1173 * In absence of 'none', 'name=' and subsystem name options,
1174 * let's default to 'all'.
1175 */
1176 if (!ctx->subsys_mask && !ctx->none && !ctx->name)
1177 ctx->all_ss = true;
1178
1179 if (ctx->all_ss) {
1180 /* Mutually exclusive option 'all' + subsystem name */
1181 if (ctx->subsys_mask)
1182 return invalfc(fc, "subsys name conflicts with all");
1183 /* 'all' => select all the subsystems */
1184 ctx->subsys_mask = enabled;
1185 }
1186
1187 /*
1188 * We either have to specify by name or by subsystems. (So all
1189 * empty hierarchies must have a name).
1190 */
1191 if (!ctx->subsys_mask && !ctx->name)
1192 return invalfc(fc, "Need name or subsystem set");
1193
1194 /*
1195 * Option noprefix was introduced just for backward compatibility
1196 * with the old cpuset, so we allow noprefix only if mounting just
1197 * the cpuset subsystem.
1198 */
1199 if ((ctx->flags & CGRP_ROOT_NOPREFIX) && (ctx->subsys_mask & mask))
1200 return invalfc(fc, "noprefix used incorrectly");
1201
1202 /* Can't specify "none" and some subsystems */
1203 if (ctx->subsys_mask && ctx->none)
1204 return invalfc(fc, "none used incorrectly");
1205
1206 return 0;
1207}

1212int cgroup1_reconfigure(struct fs_context *fc)
1213{
1214 struct cgroup_fs_context *ctx = cgroup_fc2context(fc);
1215 struct kernfs_root *kf_root = kernfs_root_from_sb(fc->root->d_sb);
1216 struct cgroup_root *root = cgroup_root_from_kf(kf_root);
1217 int ret = 0;
1218 u32 added_mask, removed_mask;
1219
1220 cgroup_lock_and_drain_offline(&cgrp_dfl_root.cgrp);
1221
1222 /* See what subsystems are wanted */
1223 ret = check_cgroupfs_options(fc);
1224 if (ret)
1225 goto out_unlock;
1226
1227 if (ctx->subsys_mask != root->subsys_mask || ctx->release_agent)
1228 pr_warn("option changes via remount are deprecated (pid=%d comm=%s)\n",
1229 task_tgid_nr(current), current->comm);
1230
1231 added_mask = ctx->subsys_mask & ~root->subsys_mask;
1232 removed_mask = root->subsys_mask & ~ctx->subsys_mask;
1233
1234 /* Don't allow flags or name to change at remount */
1235 if ((ctx->flags ^ root->flags) ||
1236 (ctx->name && strcmp(ctx->name, root->name))) {
1237 errorfc(fc, "option or name mismatch, new: 0x%x \"%s\", old: 0x%x \"%s\"",
1238 ctx->flags, ctx->name ?: "", root->flags, root->name);
1239 ret = -EINVAL;
1240 goto out_unlock;
1241 }
1242
1243 /* remounting is not allowed for populated hierarchies */
1244 if (!list_empty(&root->cgrp.self.children)) {
1245 ret = -EBUSY;
1246 goto out_unlock;
1247 }
1248
1249 ret = rebind_subsystems(root, added_mask);
1250 if (ret)
1251 goto out_unlock;
1252
1253 WARN_ON(rebind_subsystems(&cgrp_dfl_root, removed_mask));
1254
1255 if (ctx->release_agent) {
1256 spin_lock(&release_agent_path_lock);
1257 strscpy(root->release_agent_path, ctx->release_agent);
1258 spin_unlock(&release_agent_path_lock);
1259 }
1260
1261 trace_cgroup_remount(root);
1262
1263 out_unlock:
1264 cgroup_unlock();
1265 return ret;
1266}

1290static int cgroup1_root_to_use(struct fs_context *fc)
1291{
1292 struct cgroup_fs_context *ctx = cgroup_fc2context(fc);
1293 struct cgroup_root *root;
1294 struct cgroup_subsys *ss;
1295 int i, ret;
1296
1297 /* First find the desired set of subsystems */
1298 ret = check_cgroupfs_options(fc);
1299 if (ret)
1300 return ret;
1301
1302 /*
1303 * Destruction of cgroup root is asynchronous, so subsystems may
1304 * still be dying after the previous unmount. Let's drain the
1305 * dying subsystems. We just need to ensure that the ones
1306 * unmounted previously finish dying and don't care about new ones
1307 * starting. Testing ref liveliness is good enough.
1308 */
1309 for_each_subsys(ss, i) {
1310 if (!(ctx->subsys_mask & (1 << i)) ||
1311 ss->root == &cgrp_dfl_root)
1312 continue;
1313
1314 if (!percpu_ref_tryget_live(&ss->root->cgrp.self.refcnt))
1315 return 1; /* restart */
1316 cgroup_put(&ss->root->cgrp);
1317 }
1318
1319 for_each_root(root) {
1320 bool name_match = false;
1321
1322 if (root == &cgrp_dfl_root)
1323 continue;
1324
1325 /*
1326 * If we asked for a name then it must match. Also, if
1327 * name matches but sybsys_mask doesn't, we should fail.
1328 * Remember whether name matched.
1329 */
1330 if (ctx->name) {
1331 if (strcmp(ctx->name, root->name))
1332 continue;
1333 name_match = true;
1334 }
1335
1336 /*
1337 * If we asked for subsystems (or explicitly for no
1338 * subsystems) then they must match.
1339 */
1340 if ((ctx->subsys_mask || ctx->none) &&
1341 (ctx->subsys_mask != root->subsys_mask)) {
1342 if (!name_match)
1343 continue;
1344 return -EBUSY;
1345 }
1346
1347 if (root->flags ^ ctx->flags)
1348 pr_warn("new mount options do not match the existing superblock, will be ignored\n");
1349
1350 ctx->root = root;
1351 return 0;
1352 }
1353
1354 /*
1355 * No such thing, create a new one. name= matching without subsys
1356 * specification is allowed for already existing hierarchies but we
1357 * can't create new one without subsys specification.
1358 */
1359 if (!ctx->subsys_mask && !ctx->none)
1360 return invalfc(fc, "No subsys list or none specified");
1361
1362 /* Hierarchies may only be created in the initial cgroup namespace. */
1363 if (ctx->ns != &init_cgroup_ns)
1364 return -EPERM;
1365
1366 root = kzalloc_obj(*root);
1367 if (!root)
1368 return -ENOMEM;
1369
1370 ctx->root = root;
1371 init_cgroup_root(ctx);
1372
1373 ret = cgroup_setup_root(root, ctx->subsys_mask);
1374 if (!ret)
1375 cgroup_favor_dynmods(root, ctx->flags & CGRP_ROOT_FAVOR_DYNMODS);
1376 else
1377 cgroup_free_root(root);
1378
1379 return ret;
1380}

1385int cgroup1_get_tree(struct fs_context *fc)
1386{
1387 struct cgroup_fs_context *ctx = cgroup_fc2context(fc);
1388 int ret;
1389
1390 /* Check if the caller has permission to mount. */
