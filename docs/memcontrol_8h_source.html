548static inline struct mem_cgroup *folio_memcg_check(struct folio *folio)
549{
550 /*
551 * Because folio->memcg_data might be changed asynchronously
552 * for slabs, READ_ONCE() should be used here.
553 */
554 unsigned long memcg_data = READ_ONCE(folio->memcg_data);
555
556 if (memcg_data & MEMCG_DATA_OBJEXTS)
557 return NULL;
558
559 if (memcg_data & MEMCG_DATA_KMEM) {
560 struct obj_cgroup *objcg;
561
562 objcg = (void *)(memcg_data & ~OBJEXTS_FLAGS_MASK);
563 return obj_cgroup_memcg(objcg);
564 }
565
566 return (struct mem_cgroup *)(memcg_data & ~OBJEXTS_FLAGS_MASK);
567}

641static inline void mem_cgroup_protection(struct mem_cgroup *root,
642 struct mem_cgroup *memcg,
643 unsigned long *min,
644 unsigned long *low,
645 unsigned long *usage)
646{
647 *min = *low = *usage = 0;
648
649 if (mem_cgroup_disabled())
650 return;
651
652 *usage = page_counter_read(&memcg->memory);
688 if (root == memcg)
689 return;
690
691 *min = READ_ONCE(memcg->memory.emin);
692 *low = READ_ONCE(memcg->memory.elow);
693}

701static inline bool mem_cgroup_unprotected(struct mem_cgroup *target,
702 struct mem_cgroup *memcg)
703{
704 /*
705 * The root memcg doesn't account charges, and doesn't support
706 * protection. The target memcg's protection is ignored, see

710 memcg == target;

816static inline struct lruvec *mem_cgroup_lruvec(struct mem_cgroup *memcg,
817 struct pglist_data *pgdat)
818{
819 struct mem_cgroup_per_node *mz;
820 struct lruvec *lruvec;
821
822 if (mem_cgroup_disabled()) {
823 lruvec = &pgdat->__lruvec;
824 goto out;
825 }
826
827 if (!memcg)
828 memcg = root_mem_cgroup;
829
830 mz = memcg->nodeinfo[pgdat->node_id];
831 lruvec = &mz->lruvec;
832out:
833 /*
834 * Since a node can be onlined after the mem_cgroup was created,
835 * we have to be prepared to initialize lruvec->pgdat here;
836 * and if offlined then reonlined, we need to reinitialize it.
837 */
838 if (unlikely(lruvec->pgdat != pgdat))
839 lruvec->pgdat = pgdat;
840 return lruvec;
841}
