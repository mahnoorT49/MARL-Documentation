56static unsigned long cma_bitmap_aligned_mask(const struct cma *cma,
57 unsigned int align_order)
58{
59 if (align_order <= cma->order_per_bit)
60 return 0;
61 return (1UL << (align_order - cma->order_per_bit)) - 1;
62}
63
64/*
65 * Find the offset of the base PFN from the specified align_order.
66 * The value returned is represented in order_per_bits.
67 */
68static unsigned long cma_bitmap_aligned_offset(const struct cma *cma,
69 const struct cma_memrange *cmr,
70 unsigned int align_order)
71{
72 return (cmr->base_pfn & ((1UL << align_order) - 1))
73 >> cma->order_per_bit;
74}
75
76static unsigned long cma_bitmap_pages_to_bits(const struct cma *cma,
77 unsigned long pages)
78{
79 return ALIGN(pages, 1UL << cma->order_per_bit) >> cma->order_per_bit;
80}
81
82static void cma_clear_bitmap(struct cma *cma, const struct cma_memrange *cmr,
83 unsigned long pfn, unsigned long count)
84{
85 unsigned long bitmap_no, bitmap_count;
86 unsigned long flags;
87
88 bitmap_no = (pfn - cmr->base_pfn) >> cma->order_per_bit;
89 bitmap_count = cma_bitmap_pages_to_bits(cma, count);
90
91 spin_lock_irqsave(&cma->lock, flags);
92 bitmap_clear(cmr->bitmap, bitmap_no, bitmap_count);
93 cma->available_count += count;
94 spin_unlock_irqrestore(&cma->lock, flags);
95}

97/*
98 * Check if a CMA area contains no ranges that intersect with
99 * multiple zones. Store the result in the flags in case
100 * this gets called more than once.
101 */
102bool cma_validate_zones(struct cma *cma)
103{
104 int r;
105 unsigned long base_pfn;
106 struct cma_memrange *cmr;
107 bool valid_bit_set;
108
109 /*
110 * If already validated, return result of previous check.
111 * Either the valid or invalid bit will be set if this
112 * check has already been done. If neither is set, the
113 * check has not been performed yet.
114 */
115 valid_bit_set = test_bit(CMA_ZONES_VALID, &cma->flags);
116 if (valid_bit_set || test_bit(CMA_ZONES_INVALID, &cma->flags))
117 return valid_bit_set;
118
119 for (r = 0; r < cma->nranges; r++) {
120 cmr = &cma->ranges[r];
121 base_pfn = cmr->base_pfn;
122
123 /*
124 * alloc_contig_range() requires the pfn range specified
125 * to be in the same zone. Simplify by forcing the entire
126 * CMA resv range to be in the same zone.
127 */
128 WARN_ON_ONCE(!pfn_valid(base_pfn));
129 if (pfn_range_intersects_zones(cma->nid, base_pfn, cmr->count)) {
130 set_bit(CMA_ZONES_INVALID, &cma->flags);
131 return false;
132 }
133 }
134
135 set_bit(CMA_ZONES_VALID, &cma->flags);
136
137 return true;
138}

141/*
142 * cma_activate_area - Activates a reserved CMA region during early boot. It
143 * allocates bitmap tracking structures, initializes synchronization locks,
144 * marks pageblocks as MIGRATE_CMA, and sets the CMA_ACTIVATED flag. If
145 * activation fails, it rolls back bitmap allocations and safely releases the
146 * pages back to the main buddy allocator.
147 */
148static void __init cma_activate_area(struct cma *cma)
149{
150 unsigned long pfn, end_pfn, early_pfn[CMA_MAX_RANGES];
151 int allocrange, r;
152 struct cma_memrange *cmr;
153 unsigned long bitmap_count, count;
154
155 for (allocrange = 0; allocrange < cma->nranges; allocrange++) {
156 cmr = &cma->ranges[allocrange];
157 early_pfn[allocrange] = cmr->early_pfn;
158 cmr->bitmap = bitmap_zalloc(cma_bitmap_maxno(cma, cmr),
159 GFP_KERNEL);
160 if (!cmr->bitmap)
161 goto cleanup;
162 }
163
164 if (!cma_validate_zones(cma))
165 goto cleanup;
166
167 for (r = 0; r < cma->nranges; r++) {
168 cmr = &cma->ranges[r];
169 if (early_pfn[r] != cmr->base_pfn) {
170 count = early_pfn[r] - cmr->base_pfn;
171 bitmap_count = cma_bitmap_pages_to_bits(cma, count);
172 bitmap_set(cmr->bitmap, 0, bitmap_count);
173 }
174
175 for (pfn = early_pfn[r]; pfn < cmr->base_pfn + cmr->count;
176 pfn += pageblock_nr_pages)
177 init_cma_reserved_pageblock(pfn_to_page(pfn));
178 }
179
180 spin_lock_init(&cma->lock);
181
182 mutex_init(&cma->alloc_mutex);
183
184#ifdef CONFIG_CMA_DEBUGFS
185 INIT_HLIST_HEAD(&cma->mem_head);
186 spin_lock_init(&cma->mem_head_lock);
187#endif
188 set_bit(CMA_ACTIVATED, &cma->flags);
189
190 return;
191
192cleanup:
193 for (r = 0; r < allocrange; r++)
194 bitmap_free(cma->ranges[r].bitmap);
195
196 /* Expose all pages to the buddy, they are useless for CMA. */
197 if (!test_bit(CMA_RESERVE_PAGES_ON_ERROR, &cma->flags)) {
198 for (r = 0; r < allocrange; r++) {
199 cmr = &cma->ranges[r];
200 end_pfn = cmr->base_pfn + cmr->count;
201 for (pfn = early_pfn[r]; pfn < end_pfn; pfn++)
202 free_reserved_page(pfn_to_page(pfn));
203 }
204 }
205 totalcma_pages -= cma->count;
206 cma->available_count = cma->count = 0;
207 pr_err("CMA area %s could not be activated\n", cma->name);
208}

211/*
212 * cma_init_reserved_areas - A boot routine that iterates through all registered
213 * CMA regions stored in cma_areas[] and calls cma_activate_area() on each one
214 * to bring them fully online once kernel memory infrastructure is available.
215 */
216static int __init cma_init_reserved_areas(void)
217{
218 int i;
219
220 for (i = 0; i < cma_area_count; i++)
221 cma_activate_area(&cma_areas[i]);
222
223 return 0;
224}
225core_initcall(cma_init_reserved_areas);

228/*
229 * cma_reserve_pages_on_error - Sets the CMA_RESERVE_PAGES_ON_ERROR flag on a
230 * target CMA descriptor. This instructs cma_activate_area() to keep the
231 * underlying memory isolated and reserved rather than returning it to the
232 * general buddy allocator if activation fails.
233 */
234void __init cma_reserve_pages_on_error(struct cma *cma)
235{
236 set_bit(CMA_RESERVE_PAGES_ON_ERROR, &cma->flags);
237}

240/*
241 * cma_new_area - Registers a new CMA region entry in the global cma_areas[]
242 * array early in the boot sequence. It assigns the region a unique name,
243 * converts the requested byte size into page counts, records the bitmap bit
244 * granularity (order_per_bit), and increments global totalcma_pages.
245 */
246static int __init cma_new_area(const char *name, phys_addr_t size,
247 unsigned int order_per_bit,
248 struct cma **res_cma)
249{
250 struct cma *cma;
251
252 if (cma_area_count == ARRAY_SIZE(cma_areas)) {
253 pr_err("Not enough slots for CMA reserved regions!\n");
254 return -ENOSPC;
255 }
256
257 /*
258 * Each reserved area must be initialised later, when more kernel
259 * subsystems (like slab allocator) are available.
260 */
261 cma = &cma_areas[cma_area_count];
262 cma_area_count++;
263
264 if (name)
265 strscpy(cma->name, name);
266 else
267 snprintf(cma->name, CMA_MAX_NAME, "cma%d\n", cma_area_count);
268
269 cma->available_count = cma->count = size >> PAGE_SHIFT;
270 cma->order_per_bit = order_per_bit;
271 *res_cma = cma;
272 totalcma_pages += cma->count;
273
274 return 0;
275}

278/*
279 * cma_drop_area - Reverts the creation of the most recently registered region
280 * of the CMA. It subtracts the area's page count from the global totalcma_pages
281 * metric and decrements cma_area_count to safely discard an unneeded or
282 * invalid reservation.
283 */
284static void __init cma_drop_area(struct cma *cma)
285{
286 totalcma_pages -= cma->count;
287 cma_area_count--;
288}

302int __init cma_init_reserved_mem(phys_addr_t base, phys_addr_t size,
303 unsigned int order_per_bit,
304 const char *name,
305 struct cma **res_cma)
306{
307 struct cma *cma;
308 int ret;
309
310 /* Sanity checks */
311 if (!size || !memblock_is_region_reserved(base, size))
312 return -EINVAL;
313
314 /*
315 * CMA uses CMA_MIN_ALIGNMENT_BYTES as alignment requirement which
316 * needs pageblock_order to be initialized. Let's enforce it.
317 */
318 if (!pageblock_order) {
319 pr_err("pageblock_order not yet initialized. Called during early boot?\n");
320 return -EINVAL;
321 }
322
323 /* ensure minimal alignment required by mm core */
324 if (!IS_ALIGNED(base | size, CMA_MIN_ALIGNMENT_BYTES))
325 return -EINVAL;
326
327 ret = cma_new_area(name, size, order_per_bit, &cma);
328 if (ret != 0)
329 return ret;
330
331 cma->ranges[0].base_pfn = PFN_DOWN(base);
332 cma->ranges[0].early_pfn = PFN_DOWN(base);
333 cma->ranges[0].count = cma->count;
334 cma->nranges = 1;
335 cma->nid = NUMA_NO_NODE;
336
337 *res_cma = cma;
338
339 return 0;
340}

342/*
343 * Structure used while walking physical memory ranges and finding out
344 * which one(s) to use for a CMA area.

346struct cma_init_memrange {
347 phys_addr_t base;
348 phys_addr_t size;
349 struct list_head list;
350};

352/*
353 * Work array used during CMA initialization.

355static struct cma_init_memrange memranges[CMA_MAX_RANGES] __initdata;
356
357static bool __init revsizecmp(struct cma_init_memrange *mlp,
358 struct cma_init_memrange *mrp)
359{
360 return mlp->size > mrp->size;
361}
362
363static bool __init basecmp(struct cma_init_memrange *mlp,
364 struct cma_init_memrange *mrp)
365{
366 return mlp->base < mrp->base;
367}

369/*
370 * Helper function to create sorted lists.
371 */
372static void __init list_insert_sorted(
373 struct list_head *ranges,
374 struct cma_init_memrange *mrp,
375 bool (*cmp)(struct cma_init_memrange *lh, struct cma_init_memrange *rh))
376{
377 struct list_head *mp;
378 struct cma_init_memrange *mlp;
379
380 if (list_empty(ranges))
381 list_add(&mrp->list, ranges);
382 else {
383 list_for_each(mp, ranges) {
384 mlp = list_entry(mp, struct cma_init_memrange, list);
385 if (cmp(mlp, mrp))
386 break;
387 }
388 __list_add(&mrp->list, mlp->list.prev, &mlp->list);
389 }
390}

393/*
394 * cma_fixed_reserve - Checks whether a fixed physical address range crosses the
395 * highmem/lowmem boundary under CONFIG_HIGHMEM. If valid, it verifies the range
396 * is unreserved and calls memblock_reserve() to mark the physical memory as reserved.
397 */
398static int __init cma_fixed_reserve(phys_addr_t base, phys_addr_t size)
399{
400 if (IS_ENABLED(CONFIG_HIGHMEM)) {
401 phys_addr_t highmem_start = __pa(high_memory - 1) + 1;
402
403 /*
404 * If allocating at a fixed base the request region must not
405 * cross the low/high memory boundary.
406 */
407 if (base < highmem_start && base + size > highmem_start) {
408 pr_err("Region at %pa defined on low/high memory boundary (%pa)\n",
409 &base, &highmem_start);
410 return -EINVAL;
411 }
412 }
413
414 if (memblock_is_region_reserved(base, size) ||
415 memblock_reserve(base, size) < 0) {
416 return -EBUSY;
417 }
418
419 return 0;
420}

423/*
424 * cma_alloc_mem - Allocates physical memory for CMA using memblock. On 64-bit
425 * systems, it attempts bottom-up allocation above 4GB; under CONFIG_HIGHMEM,
426 * it attempts allocation in high memory first; otherwise, it calls
427 * memblock_alloc_range_nid() within the specified base and limit.
428 */
429static phys_addr_t __init cma_alloc_mem(phys_addr_t base, phys_addr_t size,
430 phys_addr_t align, phys_addr_t limit, int nid)
431{
432 phys_addr_t addr = 0;
433
434 /*
435 * If there is enough memory, try a bottom-up allocation first.
436 * It will place the new cma area close to the start of the node
437 * and guarantee that the compaction is moving pages out of the
438 * cma area and not into it.
439 * Avoid using first 4GB to not interfere with constrained zones
440 * like DMA/DMA32.
441 */
442#ifdef CONFIG_PHYS_ADDR_T_64BIT
443 if (!memblock_bottom_up() && limit >= SZ_4G + size) {
444 memblock_set_bottom_up(true);
445 addr = memblock_alloc_range_nid(size, align, SZ_4G, limit,
446 nid, true);
447 memblock_set_bottom_up(false);
448 }
449#endif
450
451 /*
452 * On systems with HIGHMEM try allocating from there before consuming
453 * memory in lower zones.
454 */
455 if (!addr && IS_ENABLED(CONFIG_HIGHMEM)) {
456 phys_addr_t highmem = __pa(high_memory - 1) + 1;
457
458 /*
459 * All pages in the reserved area must come from the same zone.
460 * If the requested region crosses the low/high memory boundary,
461 * try allocating from high memory first and fall back to low
462 * memory in case of failure.
463 */
464 if (base < highmem && limit > highmem) {
465 addr = memblock_alloc_range_nid(size, align, highmem,
466 limit, nid, true);
467 limit = highmem;
468 }
469 }
470
471 if (!addr)
472 addr = memblock_alloc_range_nid(size, align, base, limit, nid,
473 true);
474
475 return addr;
476}

479/*
480 * __cma_declare_contiguous_nid - Validates CMA configuration parameters
481 * including alignment, size, slot availability, and DRAM limits. It reserves
482 * physical memory using either cma_fixed_reserve() or cma_alloc_mem(), excludes
483 * the address range from kmemleak scanning, and registers the region via
484 * cma_init_reserved_mem() with its assigned NUMA node ID.
485 */
486static int __init __cma_declare_contiguous_nid(phys_addr_t *basep,
487 phys_addr_t size, phys_addr_t limit,
488 phys_addr_t alignment, unsigned int order_per_bit,
489 bool fixed, const char *name, struct cma **res_cma,
490 int nid)
491{
492 phys_addr_t memblock_end = memblock_end_of_DRAM();
493 phys_addr_t base = *basep;
494 int ret;
495
496 pr_debug("%s(size %pa, base %pa, limit %pa alignment %pa)\n",
497 __func__, &size, &base, &limit, &alignment);
498
499 if (cma_area_count == ARRAY_SIZE(cma_areas)) {
500 pr_err("Not enough slots for CMA reserved regions!\n");
501 return -ENOSPC;
502 }
503
504 if (!size)
505 return -EINVAL;
506
507 if (alignment && !is_power_of_2(alignment))
508 return -EINVAL;
509
510 if (!IS_ENABLED(CONFIG_NUMA))
511 nid = NUMA_NO_NODE;
512
513 /* Sanitise input arguments. */
514 alignment = max_t(phys_addr_t, alignment, CMA_MIN_ALIGNMENT_BYTES);
515 if (fixed && base & (alignment - 1)) {
516 pr_err("Region at %pa must be aligned to %pa bytes\n",
517 &base, &alignment);
518 return -EINVAL;
519 }
520 base = ALIGN(base, alignment);
521 size = ALIGN(size, alignment);
522 limit &= ~(alignment - 1);
523
524 if (!base)
525 fixed = false;
526
527 /* size should be aligned with order_per_bit */
528 if (!IS_ALIGNED(size >> PAGE_SHIFT, 1 << order_per_bit))
529 return -EINVAL;
530
531
532 /*
533 * If the limit is unspecified or above the memblock end, its effective
534 * value will be the memblock end. Set it explicitly to simplify further
535 * checks.
536 */
537 if (limit == 0 || limit > memblock_end)
538 limit = memblock_end;
539
540 if (base + size > limit) {
541 pr_err("Size (%pa) of region at %pa exceeds limit (%pa)\n",
542 &size, &base, &limit);
543 return -EINVAL;
544 }
545
546 /* Reserve memory */
547 if (fixed) {
548 ret = cma_fixed_reserve(base, size);
549 if (ret)
550 return ret;
551 } else {
552 base = cma_alloc_mem(base, size, alignment, limit, nid);
553 if (!base)
554 return -ENOMEM;
555
556 /*
557 * kmemleak scans/reads tracked objects for pointers to other
558 * objects but this address isn't mapped and accessible
559 */
560 kmemleak_ignore_phys(base);
561 }
562
563 ret = cma_init_reserved_mem(base, size, order_per_bit, name, res_cma);
564 if (ret) {
565 memblock_phys_free(base, size);
566 return ret;
567 }
568
569 (*res_cma)->nid = nid;
570 *basep = base;
571
572 return 0;
573}

575/*
576 * Create CMA areas with a total size of @total_size. A normal allocation
577 * for one area is tried first. If that fails, the biggest memblock
578 * ranges above 4G are selected, and allocated bottom up.
579 *
580 * The complexity here is not great, but this function will only be
581 * called during boot, and the lists operated on have fewer than
582 * CMA_MAX_RANGES elements (default value: 8).
583 */
584int __init cma_declare_contiguous_multi(phys_addr_t total_size,
585 phys_addr_t align, unsigned int order_per_bit,
586 const char *name, struct cma **res_cma, int nid)
587{
588 phys_addr_t start = 0, end;
589 phys_addr_t size, sizesum, sizeleft;
590 struct cma_init_memrange *mrp, *mlp, *failed;
591 struct cma_memrange *cmrp;
592 LIST_HEAD(ranges);
593 LIST_HEAD(final_ranges);
594 struct list_head *mp, *next;
595 int ret, nr = 1;
596 u64 i;
597 struct cma *cma;
598
599 /*
600 * First, try it the normal way, producing just one range.
601 */
602 ret = __cma_declare_contiguous_nid(&start, total_size, 0, align,
603 order_per_bit, false, name, res_cma, nid);
604 if (ret != -ENOMEM)
605 goto out;
606
607 /*
608 * Couldn't find one range that fits our needs, so try multiple
609 * ranges.
610 *
611 * No need to do the alignment checks here, the call to
612 * cma_declare_contiguous_nid above would have caught
613 * any issues. With the checks, we know that:
614 *
615 * - @align is a power of 2
616 * - @align is >= pageblock alignment
617 * - @size is aligned to @align and to @order_per_bit
618 *
619 * So, as long as we create ranges that have a base
620 * aligned to @align, and a size that is aligned to
621 * both @align and @order_to_bit, things will work out.
622 */
623 nr = 0;
624 sizesum = 0;
625 failed = NULL;
626
627 ret = cma_new_area(name, total_size, order_per_bit, &cma);
628 if (ret != 0)
629 goto out;
630
631 align = max_t(phys_addr_t, align, CMA_MIN_ALIGNMENT_BYTES);
632 /*
633 * Create a list of ranges above 4G, largest range first.
634 */
635 for_each_free_mem_range(i, nid, MEMBLOCK_NONE, &start, &end, NULL) {
636 if (upper_32_bits(start) == 0)
637 continue;
638
639 start = ALIGN(start, align);
640 if (start >= end)
641 continue;
642
643 end = ALIGN_DOWN(end, align);
644 if (end <= start)
645 continue;
646
647 size = end - start;
648 size = ALIGN_DOWN(size, (PAGE_SIZE << order_per_bit));
649 if (!size)
650 continue;
651 sizesum += size;
652
653 pr_debug("consider %016llx - %016llx\n", (u64)start, (u64)end);
654
655 /*
656 * If we don't yet have used the maximum number of
657 * areas, grab a new one.
658 *
659 * If we can't use anymore, see if this range is not
660 * smaller than the smallest one already recorded. If
661 * not, re-use the smallest element.
662 */
663 if (nr < CMA_MAX_RANGES)
664 mrp = &memranges[nr++];
665 else {
666 mrp = list_last_entry(&ranges,
667 struct cma_init_memrange, list);
668 if (size < mrp->size)
669 continue;
670 list_del(&mrp->list);
671 sizesum -= mrp->size;
672 pr_debug("deleted %016llx - %016llx from the list\n",
673 (u64)mrp->base, (u64)mrp->base + size);
674 }
675 mrp->base = start;
676 mrp->size = size;
677
678 /*
679 * Now do a sorted insert.
680 */
681 list_insert_sorted(&ranges, mrp, revsizecmp);
682 pr_debug("added %016llx - %016llx to the list\n",
683 (u64)mrp->base, (u64)mrp->base + size);
684 pr_debug("total size now %llu\n", (u64)sizesum);
685 }
686
687 /*
688 * There is not enough room in the CMA_MAX_RANGES largest
689 * ranges, so bail out.
690 */
691 if (sizesum < total_size) {
692 cma_drop_area(cma);
693 ret = -ENOMEM;
694 goto out;
695 }
696
697 /*
698 * Found ranges that provide enough combined space.
699 * Now, sorted them by address, smallest first, because we
700 * want to mimic a bottom-up memblock allocation.
701 */
702 sizesum = 0;
703 list_for_each_safe(mp, next, &ranges) {
704 mlp = list_entry(mp, struct cma_init_memrange, list);
705 list_del(mp);
706 list_insert_sorted(&final_ranges, mlp, basecmp);
707 sizesum += mlp->size;
708 if (sizesum >= total_size)
709 break;
710 }
711
712 /*
713 * Walk the final list, and add a CMA range for
714 * each range, possibly not using the last one fully.
715 */
716 nr = 0;
717 sizeleft = total_size;
718 list_for_each(mp, &final_ranges) {
719 mlp = list_entry(mp, struct cma_init_memrange, list);
720 size = min(sizeleft, mlp->size);
721 if (memblock_reserve(mlp->base, size)) {
722 /*
723 * Unexpected error. Could go on to
724 * the next one, but just abort to
725 * be safe.
726 */
727 failed = mlp;
728 break;
729 }
730
731 pr_debug("created region %d: %016llx - %016llx\n",
732 nr, (u64)mlp->base, (u64)mlp->base + size);
733 cmrp = &cma->ranges[nr++];
734 cmrp->base_pfn = PHYS_PFN(mlp->base);
735 cmrp->early_pfn = cmrp->base_pfn;
736 cmrp->count = size >> PAGE_SHIFT;
737
738 sizeleft -= size;
739 if (sizeleft == 0)
740 break;
741 }
742
743 if (failed) {
744 list_for_each(mp, &final_ranges) {
745 mlp = list_entry(mp, struct cma_init_memrange, list);
746 if (mlp == failed)
747 break;
748 memblock_phys_free(mlp->base, mlp->size);
749 }
750 cma_drop_area(cma);
751 ret = -ENOMEM;
752 goto out;
753 }
754
755 cma->nranges = nr;
756 cma->nid = nid;
757 *res_cma = cma;
758
759out:
760 if (ret != 0)
761 pr_err("Failed to reserve %lu MiB\n",
762 (unsigned long)total_size / SZ_1M);
763 else
764 pr_info("Reserved %lu MiB in %d range%s\n",
765 (unsigned long)total_size / SZ_1M, nr, str_plural(nr));
766 return ret;
767}

844/*
845 * cma_range_alloc - Searches a specific memory range bitmap for an aligned
846 * sequence of zero bits matching the requested page count. After marking the
847 * matching bits as allocated and decrementing available_count, it releases the
848 * spinlock and calls alloc_contig_frozen_range() to migrate existing pages out
849 * of the span, clearing the bitmap and retrying if migration fails.
850 */
851static int cma_range_alloc(struct cma *cma, struct cma_memrange *cmr,
852 unsigned long count, unsigned int align,
853 struct page **pagep, gfp_t gfp)
854{
855 unsigned long bitmap_maxno, bitmap_no, bitmap_count;
856 unsigned long start, pfn, mask, offset;
857 int ret = -EBUSY;
858 struct page *page = NULL;
859
860 mask = cma_bitmap_aligned_mask(cma, align);
861 offset = cma_bitmap_aligned_offset(cma, cmr, align);
862 bitmap_maxno = cma_bitmap_maxno(cma, cmr);
863 bitmap_count = cma_bitmap_pages_to_bits(cma, count);
864
865 if (bitmap_count > bitmap_maxno)
866 goto out;
867
868 for (start = 0; ; start = bitmap_no + mask + 1) {
869 spin_lock_irq(&cma->lock);
870 /*
871 * If the request is larger than the available number
872 * of pages, stop right away.
873 */
874 if (count > cma->available_count) {
875 spin_unlock_irq(&cma->lock);
876 break;
877 }
878 bitmap_no = bitmap_find_next_zero_area_off(cmr->bitmap,
879 bitmap_maxno, start, bitmap_count, mask,
880 offset);
881 if (bitmap_no >= bitmap_maxno) {
882 spin_unlock_irq(&cma->lock);
883 break;
884 }
885
886 pfn = cmr->base_pfn + (bitmap_no << cma->order_per_bit);
887 page = pfn_to_page(pfn);
888
889 /*
890 * Do not hand out page ranges that are not contiguous, so
891 * callers can just iterate the pages without having to worry
892 * about these corner cases.
893 */
894 if (!page_range_contiguous(page, count)) {
895 spin_unlock_irq(&cma->lock);
896 pr_warn_ratelimited("%s: %s: skipping incompatible area [0x%lx-0x%lx]",
897 __func__, cma->name, pfn, pfn + count - 1);
898 continue;
899 }
900
901 bitmap_set(cmr->bitmap, bitmap_no, bitmap_count);
902 cma->available_count -= count;
903 /*
904 * It's safe to drop the lock here. We've marked this region for
905 * our exclusive use. If the migration fails we will take the
906 * lock again and unmark it.
907 */
908 spin_unlock_irq(&cma->lock);
909
910 mutex_lock(&cma->alloc_mutex);
911 ret = alloc_contig_frozen_range(pfn, pfn + count, ACR_FLAGS_CMA, gfp);
912 mutex_unlock(&cma->alloc_mutex);
913 if (!ret)
914 break;
915
916 cma_clear_bitmap(cma, cmr, pfn, count);
917 if (ret != -EBUSY)
918 break;
919
920 pr_debug("%s(): memory range at pfn 0x%lx %p is busy, retrying\n",
921 __func__, pfn, page);
922
923 trace_cma_alloc_busy_retry(cma->name, pfn, page, count, align);
924 }
925out:
926 if (!ret)
927 *pagep = page;
928 return ret;
929}

932/*
933 * __cma_alloc_frozen - Iterates through all memory ranges within a CMA area to
934 * allocate contiguous pages via cma_range_alloc(). On success, it resets KASAN
935 * memory tags across the allocated page range and increments success metrics.
936 * On failure, it records failure statistics and invokes cma_debug_show_areas()
937 * if __GFP_NOWARN is absent.
938 */
939static struct page *__cma_alloc_frozen(struct cma *cma,
940 unsigned long count, unsigned int align, gfp_t gfp)
941{
942 struct page *page = NULL;
943 int ret = -ENOMEM, r;
944 unsigned long i;
945 const char *name = cma ? cma->name : NULL;
946
947 if (!cma || !cma->count)
948 return page;
949
950 pr_debug("%s(cma %p, name: %s, count %lu, align %d)\n", __func__,
951 (void *)cma, cma->name, count, align);
952
953 if (!count)
954 return page;
955
956 trace_cma_alloc_start(name, count, cma->available_count, cma->count, align);
957
958 for (r = 0; r < cma->nranges; r++) {
959 page = NULL;
960
961 ret = cma_range_alloc(cma, &cma->ranges[r], count, align,
962 &page, gfp);
963 if (ret != -EBUSY || page)
964 break;
965 }
966
967 /*
968 * CMA can allocate multiple page blocks, which results in different
969 * blocks being marked with different tags. Reset the tags to ignore
970 * those page blocks.
971 */
972 if (page) {
973 for (i = 0; i < count; i++)
974 page_kasan_tag_reset(page + i);
975 }
976
977 if (ret && !(gfp & __GFP_NOWARN)) {
978 pr_err_ratelimited("%s: %s: alloc failed, req-size: %lu pages, ret: %d\n",
979 __func__, cma->name, count, ret);
980 cma_debug_show_areas(cma);
981 }
982
983 pr_debug("%s(): returned %p\n", __func__, page);
984 trace_cma_alloc_finish(name, page ? page_to_pfn(page) : 0,
985 page, count, align, ret);
986 if (page) {
987 count_vm_event(CMA_ALLOC_SUCCESS);
988 cma_sysfs_account_success_pages(cma, count);
989 } else {
990 count_vm_event(CMA_ALLOC_FAIL);
991 cma_sysfs_account_fail_pages(cma, count);
992 }
993
994 return page;
995}

1036 * find_cma_memrange - Maps a physical page range back to its containing
1037 * cma_memrange structure within a CMA area. It converts the pages pointer to a
1038 * PFN, verifies that the entire span (pfn through pfn + count) falls within a
1039 * single range boundary, and returns the corresponding cma_memrange pointer or
1040 * NULL if no valid match exists.
1041 */

1043 const struct page *pages, unsigned long count)
1044{
1045 struct cma_memrange *cmr = NULL;
1046 unsigned long pfn, end_pfn;

1050

1052 return NULL;

1054 pfn = page_to_pfn(pages);
1055
1056 for (r = 0; r < cma->nranges; r++) {
1057 cmr = &cma->ranges[r];

1059 if (pfn >= cmr->base_pfn && pfn < end_pfn) {
1060 if (pfn + count <= end_pfn)
1061 break;
1062
1063 VM_WARN_ON_ONCE(1);

1065 }
1066
1067 if (r == cma->nranges) {
1068 pr_debug("%s(page %p, count %lu, no cma range matches the page range)\n",
1069 __func__, (void *)pages, count);

1071 }
1072
1073 return cmr;

1198/*
1199 * Very basic function to reserve memory from a CMA area that has not
1200 * yet been activated. This is expected to be called early, when the
1201 * system is single-threaded, so there is no locking. The alignment
1202 * checking is restrictive - only pageblock-aligned areas
1203 * (CMA_MIN_ALIGNMENT_BYTES) may be reserved through this function.
1204 * This keeps things simple, and is enough for the current use case.
1205 *
1206 * The CMA bitmaps have not yet been allocated, so just start
1207 * reserving from the bottom up, using a PFN to keep track
1208 * of what has been reserved. Unreserving is not possible.
1209 *
1210 * The caller is responsible for initializing the page structures
1211 * in the area properly, since this just points to memblock-allocated
1212 * memory. The caller should subsequently use init_cma_pageblock to
1213 * set the migrate type and CMA stats the pageblocks that were reserved.
1214 *
1215 * If the CMA area fails to activate later, memory obtained through
1216 * this interface is not handed to the page allocator, this is
1217 * the responsibility of the caller (e.g. like normal memblock-allocated
1218 * memory).
1219 */
1220void __init *cma_reserve_early(struct cma *cma, unsigned long size)
1221{
1222 int r;
1223 struct cma_memrange *cmr;
1224 unsigned long available;
1225 void *ret = NULL;
1226
1227 if (!cma || !cma->count)
1228 return NULL;
1229 /*
1230 * Can only be called early in init.
1231 */
1232 if (test_bit(CMA_ACTIVATED, &cma->flags))
1233 return NULL;
1234
1235 if (!IS_ALIGNED(size, CMA_MIN_ALIGNMENT_BYTES))
1236 return NULL;
1237
1238 if (!IS_ALIGNED(size, (PAGE_SIZE << cma->order_per_bit)))
1239 return NULL;
1240
1241 size >>= PAGE_SHIFT;
1242
1243 if (size > cma->available_count)
1244 return NULL;
1245
1246 for (r = 0; r < cma->nranges; r++) {
1247 cmr = &cma->ranges[r];
1248 available = cmr->count - (cmr->early_pfn - cmr->base_pfn);
1249 if (size <= available) {
1250 ret = phys_to_virt(PFN_PHYS(cmr->early_pfn));
1251 cmr->early_pfn += size;
1252 cma->available_count -= size;
1253 return ret;
1254 }
1255 }
1256
1257 return ret;
1258}
