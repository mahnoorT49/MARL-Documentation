2/*
3 * DMA Pool allocator
4 *
5 * Copyright 2001 David Brownell
6 * Copyright 2007 Intel Corporation
7 * Author: Matthew Wilcox <willy@linux.intel.com>
8 *
9 * This allocator returns small blocks of a given size which are DMA-able by
10 * the given device. It uses the dma_alloc_coherent page allocator to get
11 * new pages, then splits them up into blocks of the required size.
12 * Many older drivers still have their own code to do this.
13 *
14 * The current design of this allocator is fairly simple. The pool is
15 * represented by the 'struct dma_pool' which keeps a doubly-linked list of
16 * allocated pages. Each page in the page_list is split into blocks of at
17 * least 'size' bytes. Free blocks are tracked in an unsorted singly-linked
18 * list of free blocks across all pages. Used blocks aren't tracked, but we
19 * keep a count of how many are currently allocated from each page.
20 */
21

111static void pool_check_block(struct dma_pool *pool, struct dma_block *block,
112 gfp_t mem_flags)
113{
114 u8 *data = (void *)block;
115 int i;
116
117 for (i = sizeof(struct dma_block); i < pool->size; i++) {
118 if (data[i] == POOL_POISON_FREED)
119 continue;
120 dev_err(pool->dev, "%s %s, %p (corrupted)\n", __func__,
121 pool->name, block);
122
123 /*
124 * Dump the first 4 bytes even if they are not
125 * POOL_POISON_FREED
126 */
127 print_hex_dump(KERN_ERR, "", DUMP_PREFIX_OFFSET, 16, 1,
128 data, pool->size, 1);
129 break;
130 }
131
132 if (!want_init_on_alloc(mem_flags))
133 memset(block, POOL_POISON_ALLOCATED, pool->size);
134}

139static struct dma_page *pool_find_page(struct dma_pool *pool, dma_addr_t dma)
140{
141 struct dma_page *page;
142
143 list_for_each_entry(page, &pool->page_list, page_list) {
144 if (dma < page->dma)
145 continue;
146 if ((dma - page->dma) < pool->allocation)
147 return page;
148 }
149 return NULL;
150}

155static bool pool_block_err(struct dma_pool *pool, void *vaddr, dma_addr_t dma)
156{
157 struct dma_block *block = pool->next_block;
158 struct dma_page *page;
159
160 page = pool_find_page(pool, dma);
161 if (!page) {
162 dev_err(pool->dev, "%s %s, %p/%pad (bad dma)\n",
163 __func__, pool->name, vaddr, &dma);
164 return true;
165 }
166
167 while (block) {
168 if (block != vaddr) {
169 block = block->next_block;
170 continue;
171 }
172 dev_err(pool->dev, "%s %s, dma %pad already free\n",
173 __func__, pool->name, &dma);
174 return true;
175 }
176
177 memset(vaddr, POOL_POISON_FREED, pool->size);
178 return false;
179}

209static struct dma_block *pool_block_pop(struct dma_pool *pool)
210{
211 struct dma_block *block = pool->next_block;
212
213 if (block) {
214 pool->next_block = block->next_block;
215 pool->nr_active++;
216 }
217 return block;
218}

260struct dma_pool *dma_pool_create_node(const char *name, struct device *dev,
261 size_t size, size_t align, size_t boundary, int node)
262{
263 struct dma_pool *retval;
264 size_t allocation;
265 bool empty;
266
267 if (!dev)
268 return NULL;
269
270 if (align == 0)
271 align = 1;
272 else if (align & (align - 1))
273 return NULL;
274
275 if (size == 0 || size > INT_MAX)
276 return NULL;
277 if (size < sizeof(struct dma_block))
278 size = sizeof(struct dma_block);
279
280 size = ALIGN(size, align);
281 allocation = max_t(size_t, size, PAGE_SIZE);
282
283 if (!boundary)
284 boundary = allocation;
285 else if ((boundary < size) || (boundary & (boundary - 1)))
286 return NULL;
287
288 boundary = min(boundary, allocation);
289
290 retval = kzalloc_node(sizeof(*retval), GFP_KERNEL, node);
291 if (!retval)
292 return retval;
293
294 strscpy(retval->name, name, sizeof(retval->name));
295
296 retval->dev = dev;
297
298 INIT_LIST_HEAD(&retval->page_list);
299 spin_lock_init(&retval->lock);
300 retval->size = size;
301 retval->boundary = boundary;
302 retval->allocation = allocation;
303 retval->node = node;
304 INIT_LIST_HEAD(&retval->pools);
305
306 /*
307 * pools_lock ensures that the ->dma_pools list does not get corrupted.
308 * pools_reg_lock ensures that there is not a race between
309 * dma_pool_create() and dma_pool_destroy() or within dma_pool_create()
310 * when the first invocation of dma_pool_create() failed on
311 * device_create_file() and the second assumes that it has been done (I
312 * know it is a short window).
313 */
314 mutex_lock(&pools_reg_lock);
315 mutex_lock(&pools_lock);
316 empty = list_empty(&dev->dma_pools);
317 list_add(&retval->pools, &dev->dma_pools);
318 mutex_unlock(&pools_lock);
319 if (empty) {
320 int err;
321
322 err = device_create_file(dev, &dev_attr_pools);
323 if (err) {
324 mutex_lock(&pools_lock);
325 list_del(&retval->pools);
326 mutex_unlock(&pools_lock);
327 mutex_unlock(&pools_reg_lock);
328 kfree(retval);
329 return NULL;
330 }
331 }
332 mutex_unlock(&pools_reg_lock);
333 return retval;
334}
335EXPORT_SYMBOL(dma_pool_create_node);

340static void pool_initialise_page(struct dma_pool *pool, struct dma_page *page)
341{
342 unsigned int next_boundary = pool->boundary, offset = 0;
343 struct dma_block *block, *first = NULL, *last = NULL;
344
345 pool_init_page(pool, page);
346 while (offset + pool->size <= pool->allocation) {
347 if (offset + pool->size > next_boundary) {
348 offset = next_boundary;
349 next_boundary += pool->boundary;
350 continue;
351 }
352
353 block = page->vaddr + offset;
354 block->dma = page->dma + offset;
355 block->next_block = NULL;
356
357 if (last)
358 last->next_block = block;
359 else
360 first = block;
361 last = block;
362
363 offset += pool->size;
364 pool->nr_blocks++;
365 }
366
367 last->next_block = pool->next_block;
368 pool->next_block = first;
369
370 list_add(&page->page_list, &pool->page_list);
371 pool->nr_pages++;
372}

377static struct dma_page *pool_alloc_page(struct dma_pool *pool, gfp_t mem_flags)
378{
379 struct dma_page *page;
380
381 page = kmalloc_node(sizeof(*page), mem_flags, pool->node);
382 if (!page)
383 return NULL;
384
385 page->vaddr = dma_alloc_coherent(pool->dev, pool->allocation,
386 &page->dma, mem_flags);
387 if (!page->vaddr) {
388 kfree(page);
389 return NULL;
390 }
391
392 return page;
393}

406void dma_pool_destroy(struct dma_pool *pool)
407{
408 struct dma_page *page, *tmp;
409 bool empty, busy = false;
410
411 if (unlikely(!pool))
412 return;
413
414 mutex_lock(&pools_reg_lock);
415 mutex_lock(&pools_lock);
416 list_del(&pool->pools);
417 empty = list_empty(&pool->dev->dma_pools);
418 mutex_unlock(&pools_lock);
419 if (empty)
420 device_remove_file(pool->dev, &dev_attr_pools);
421 mutex_unlock(&pools_reg_lock);
422
423 if (pool->nr_active) {
424 dev_err(pool->dev, "%s %s busy\n", __func__, pool->name);
425 busy = true;
426 }
427
428 list_for_each_entry_safe(page, tmp, &pool->page_list, page_list) {
429 if (!busy)
430 dma_free_coherent(pool->dev, pool->allocation,
431 page->vaddr, page->dma);
432 list_del(&page->page_list);
433 kfree(page);
434 }
435
436 kfree(pool);
437}
438EXPORT_SYMBOL(dma_pool_destroy);

453void *dma_pool_alloc(struct dma_pool *pool, gfp_t mem_flags,
454 dma_addr_t *handle)
455{
456 struct dma_block *block;
457 struct dma_page *page;
458 unsigned long flags;
459
460 might_alloc(mem_flags);
461
462 spin_lock_irqsave(&pool->lock, flags);
463 block = pool_block_pop(pool);
464 if (!block) {
465 /*
466 * pool_alloc_page() might sleep, so temporarily drop
467 * &pool->lock
468 */
469 spin_unlock_irqrestore(&pool->lock, flags);
470
471 page = pool_alloc_page(pool, mem_flags & (~__GFP_ZERO));
472 if (!page)
473 return NULL;
474
475 spin_lock_irqsave(&pool->lock, flags);
476 pool_initialise_page(pool, page);
477 block = pool_block_pop(pool);
478 }
479 spin_unlock_irqrestore(&pool->lock, flags);
480
481 *handle = block->dma;
482 pool_check_block(pool, block, mem_flags);
483 if (want_init_on_alloc(mem_flags))
484 memset(block, 0, pool->size);
485
486 return block;
487}
488EXPORT_SYMBOL(dma_pool_alloc);

502void dma_pool_free(struct dma_pool *pool, void *vaddr, dma_addr_t dma)
503{
504 struct dma_block *block = vaddr;
505 unsigned long flags;
506
507 spin_lock_irqsave(&pool->lock, flags);
508 if (!pool_block_err(pool, vaddr, dma)) {
509 pool_block_push(pool, block, dma);
510 pool->nr_active--;
511 }
512 spin_unlock_irqrestore(&pool->lock, flags);
513}
514EXPORT_SYMBOL(dma_pool_free);
