349static void zspage_read_lock(struct zspage *zspage)
350{
351 struct zspage_lock *zsl = &zspage->zsl;
352
353 rwsem_acquire_read(&zsl->dep_map, 0, 0, _RET_IP_);
354
355 spin_lock(&zsl->lock);
356 zsl->cnt++;
357 spin_unlock(&zsl->lock);
358
359 lock_acquired(&zsl->dep_map, _RET_IP_);
360}

365static void zspage_read_unlock(struct zspage *zspage)
366{
367 struct zspage_lock *zsl = &zspage->zsl;
368
369 rwsem_release(&zsl->dep_map, _RET_IP_);
370
371 spin_lock(&zsl->lock);
372 zsl->cnt--;
373 spin_unlock(&zsl->lock);
374}

379static __must_check bool zspage_write_trylock(struct zspage *zspage)
380{
381 struct zspage_lock *zsl = &zspage->zsl;
382
383 spin_lock(&zsl->lock);
384 if (zsl->cnt == ZS_PAGE_UNLOCKED) {
385 zsl->cnt = ZS_PAGE_WRLOCKED;
386 rwsem_acquire(&zsl->dep_map, 0, 1, _RET_IP_);
387 lock_acquired(&zsl->dep_map, _RET_IP_);
388 return true;
389 }
390
391 spin_unlock(&zsl->lock);
392 return false;
393}

398static void zspage_write_unlock(struct zspage *zspage)
399{
400 struct zspage_lock *zsl = &zspage->zsl;
401
402 rwsem_release(&zsl->dep_map, _RET_IP_);
403
404 zsl->cnt = ZS_PAGE_UNLOCKED;
405 spin_unlock(&zsl->lock);
406}

638static int zs_stats_size_show(struct seq_file *s, void *v)
639{
640 int i, fg;
641 struct zs_pool *pool = s->private;
642 struct size_class *class;
643 int objs_per_zspage;
644 unsigned long obj_allocated, obj_used, pages_used, freeable;
645 unsigned long total_objs = 0, total_used_objs = 0, total_pages = 0;
646 unsigned long total_freeable = 0;
647 unsigned long inuse_totals[NR_FULLNESS_GROUPS] = {0, };
648
649 seq_printf(s, " %5s %5s %9s %9s %9s %9s %9s %9s %9s %9s %9s %9s %9s %13s %10s %10s %16s %8s\n",
650 "class", "size", "10%", "20%", "30%", "40%",
651 "50%", "60%", "70%", "80%", "90%", "99%", "100%",
652 "obj_allocated", "obj_used", "pages_used",
653 "pages_per_zspage", "freeable");
654
655 for (i = 0; i < ZS_SIZE_CLASSES; i++) {
656
657 class = pool->size_class[i];
658
659 if (class->index != i)
660 continue;
661
662 spin_lock(&class->lock);
663
664 seq_printf(s, " %5u %5u ", i, class->size);
665 for (fg = ZS_INUSE_RATIO_10; fg < NR_FULLNESS_GROUPS; fg++) {
666 inuse_totals[fg] += class_stat_read(class, fg);
667 seq_printf(s, "%9lu ", class_stat_read(class, fg));
668 }
669
670 obj_allocated = class_stat_read(class, ZS_OBJS_ALLOCATED);
671 obj_used = class_stat_read(class, ZS_OBJS_INUSE);
672 freeable = zs_can_compact(class);
673 spin_unlock(&class->lock);
674
675 objs_per_zspage = class->objs_per_zspage;
676 pages_used = obj_allocated / objs_per_zspage *
677 class->pages_per_zspage;
678
679 seq_printf(s, "%13lu %10lu %10lu %16d %8lu\n",
680 obj_allocated, obj_used, pages_used,
681 class->pages_per_zspage, freeable);
682
683 total_objs += obj_allocated;
684 total_used_objs += obj_used;
685 total_pages += pages_used;
686 total_freeable += freeable;
687 }
688
689 seq_puts(s, "\n");
690 seq_printf(s, " %5s %5s ", "Total", "");
691
692 for (fg = ZS_INUSE_RATIO_10; fg < NR_FULLNESS_GROUPS; fg++)
693 seq_printf(s, "%9lu ", inuse_totals[fg]);
694
695 seq_printf(s, "%13lu %10lu %10lu %16s %8lu\n",
696 total_objs, total_used_objs, total_pages, "",
697 total_freeable);
698
699 return 0;
700}

754static int get_fullness_group(struct size_class *class, struct zspage *zspage)
755{
756 int inuse, objs_per_zspage, ratio;
757
758 inuse = get_zspage_inuse(zspage);
759 objs_per_zspage = class->objs_per_zspage;
760
761 if (inuse == 0)
762 return ZS_INUSE_RATIO_0;
763 if (inuse == objs_per_zspage)
764 return ZS_INUSE_RATIO_100;
765
766 ratio = 100 * inuse / objs_per_zspage;
767 /*
768 * Take integer division into consideration: a page with one inuse
769 * object out of 127 possible, will end up having 0 usage ratio,
770 * which is wrong as it belongs in ZS_INUSE_RATIO_10 fullness group.
771 */
772 return ratio / 10 + 1;
773}

913static inline bool obj_allocated(struct zpdesc *zpdesc, void *obj,
914 unsigned long *phandle)
915{
916 unsigned long handle;
917 struct zspage *zspage = get_zspage(zpdesc);

920 VM_BUG_ON_PAGE(!is_first_zpdesc(zpdesc), zpdesc_page(zpdesc));
921 handle = zpdesc->handle;

926 return false;
927
928 /* Clear all tags before returning the handle */
929 *phandle = handle & ~OBJ_TAG_MASK;
930 return true;

949static int trylock_zspage(struct zspage *zspage)
950{
951 struct zpdesc *cursor, *fail;
952
953 for (cursor = get_first_zpdesc(zspage); cursor != NULL; cursor =
954 get_next_zpdesc(cursor)) {
955 if (!zpdesc_trylock(cursor)) {
956 fail = cursor;
957 goto unlock;
958 }
959 }
960
961 return 1;
962unlock:
963 for (cursor = get_first_zpdesc(zspage); cursor != fail; cursor =
964 get_next_zpdesc(cursor))
965 zpdesc_unlock(cursor);
966
967 return 0;
968}

973static void __free_zspage(struct zs_pool *pool, struct size_class *class,
974 struct zspage *zspage)
975{
976 struct zpdesc *zpdesc, *next;
977
978 assert_spin_locked(&class->lock);
979
980 VM_BUG_ON(get_zspage_inuse(zspage));
981 VM_BUG_ON(zspage->fullness != ZS_INUSE_RATIO_0);
982
983 next = zpdesc = get_first_zpdesc(zspage);
984 do {
985 VM_BUG_ON_PAGE(!zpdesc_is_locked(zpdesc), zpdesc_page(zpdesc));
986 next = get_next_zpdesc(zpdesc);
987 reset_zpdesc(zpdesc);
988 zpdesc_unlock(zpdesc);
989 zpdesc_dec_zone_page_state(zpdesc);
990 zpdesc_put(zpdesc);
991 zpdesc = next;
992 } while (zpdesc != NULL);
993
994 cache_free_zspage(zspage);
995
996 class_stat_sub(class, ZS_OBJS_ALLOCATED, class->objs_per_zspage);
997 atomic_long_sub(class->pages_per_zspage, &pool->pages_allocated);
998}

1003static void free_zspage(struct zs_pool *pool, struct size_class *class,
1004 struct zspage *zspage)
1005{
1006 VM_BUG_ON(get_zspage_inuse(zspage));
1007 VM_BUG_ON(list_empty(&zspage->list));
1008
1009 /*
1010 * Since zs_free couldn't be sleepable, this function cannot call
1011 * lock_page. The page locks trylock_zspage got will be released
1012 * by __free_zspage.
1013 */
1014 if (!trylock_zspage(zspage)) {
1015 kick_deferred_free(pool);
1016 return;
1017 }
1018
1019 remove_zspage(class, zspage);
1020 __free_zspage(pool, class, zspage);
1021}

1028static void init_zspage(struct size_class *class, struct zspage *zspage)
1029{
1030 unsigned int freeobj = 1;
1031 unsigned long off = 0;
1032 struct zpdesc *zpdesc = get_first_zpdesc(zspage);
1033
1034 while (zpdesc) {
1035 struct zpdesc *next_zpdesc;
1036 struct link_free *link;
1037 void *vaddr;
1038
1039 set_first_obj_offset(zpdesc, off);
1040
1041 vaddr = kmap_local_zpdesc(zpdesc);
1042 link = (struct link_free *)vaddr + off / sizeof(*link);
1043
1044 while ((off += class->size) < PAGE_SIZE) {
1045 link->next = freeobj++ << OBJ_TAG_BITS;
1046 link += class->size / sizeof(*link);
1047 }
1048
1049 /*
1050 * We now come to the last (full or partial) object on this
1051 * page, which must point to the first object on the next
1052 * page (if present)
1053 */
1054 next_zpdesc = get_next_zpdesc(zpdesc);
1055 if (next_zpdesc) {
1056 link->next = freeobj++ << OBJ_TAG_BITS;
1057 } else {
1058 /*
1059 * Reset OBJ_TAG_BITS bit to last link to tell
1060 * whether it's allocated object or not.
1061 */
1062 link->next = -1UL << OBJ_TAG_BITS;
1063 }
1064 kunmap_local(vaddr);
1065 zpdesc = next_zpdesc;
1066 off %= PAGE_SIZE;
1067 }
1068
1069 set_freeobj(zspage, 0);
1070}

1075static void create_page_chain(struct size_class *class, struct zspage *zspage,
1076 struct zpdesc *zpdescs[])
1077{
1078 int i;
1079 struct zpdesc *zpdesc;
1080 struct zpdesc *prev_zpdesc = NULL;
1081 int nr_zpdescs = class->pages_per_zspage;
1082
1083 /*
1084 * Allocate individual pages and link them together as:
1085 * 1. all pages are linked together using zpdesc->next
1086 * 2. each sub-page point to zspage using zpdesc->zspage
1087 *
1088 * we set PG_private to identify the first zpdesc (i.e. no other zpdesc
1089 * has this flag set).
1090 */
1091 for (i = 0; i < nr_zpdescs; i++) {
1092 zpdesc = zpdescs[i];
1093 zpdesc->zspage = zspage;
1094 zpdesc->next = NULL;
1095 if (i == 0) {
1096 zspage->first_zpdesc = zpdesc;
1097 zpdesc_set_first(zpdesc);
1098 if (unlikely(class->objs_per_zspage == 1 &&
1099 class->pages_per_zspage == 1))
1100 SetZsHugePage(zspage);
1101 } else {
1102 prev_zpdesc->next = zpdesc;
1103 }
1104 prev_zpdesc = zpdesc;
1105 }
1106}

1114static struct zspage *alloc_zspage(struct zs_pool *pool,
1115 struct size_class *class,
1116 gfp_t gfp, const int nid)
1117{
1118 int i;
1119 struct zpdesc *zpdescs[ZS_MAX_PAGES_PER_ZSPAGE];
1120 struct zspage *zspage = cache_alloc_zspage(gfp);
1121
1122 if (!zspage)
1123 return NULL;
1124
1125 if (!IS_ENABLED(CONFIG_COMPACTION))
1126 gfp &= ~__GFP_MOVABLE;
1127
1128 zspage->magic = ZSPAGE_MAGIC;
1129 zspage->pool = pool;
1130 zspage->class = class->index;
1131 zspage_lock_init(zspage);
1132
1133 for (i = 0; i < class->pages_per_zspage; i++) {
1134 struct zpdesc *zpdesc;
1135
1136 zpdesc = alloc_zpdesc(gfp, nid);
1137 if (!zpdesc) {
1138 while (--i >= 0) {
1139 zpdesc_dec_zone_page_state(zpdescs[i]);
1140 free_zpdesc(zpdescs[i]);
1141 }
1142 cache_free_zspage(zspage);
1143 return NULL;
1144 }
1145 __zpdesc_set_zsmalloc(zpdesc);
1146
1147 zpdesc_inc_zone_page_state(zpdesc);
1148 zpdescs[i] = zpdesc;
1149 }
1150
1151 create_page_chain(class, zspage, zpdescs);
1152 init_zspage(class, zspage);
1153
1154 return zspage;

1465static unsigned long obj_malloc(struct zs_pool *pool,
1466 struct zspage *zspage, unsigned long handle)
1467{
1468 int i, nr_zpdesc, offset;
1469 unsigned long obj;
1470 struct link_free *link;
1471 struct size_class *class;
1472
1473 struct zpdesc *m_zpdesc;
1474 unsigned long m_offset;
1475 void *vaddr;
1476
1477 class = pool->size_class[zspage->class];
1478 obj = get_freeobj(zspage);
1479
1480 offset = obj * class->size;
1481 nr_zpdesc = offset >> PAGE_SHIFT;
1482 m_offset = offset_in_page(offset);
1483 m_zpdesc = get_first_zpdesc(zspage);
1484
1485 for (i = 0; i < nr_zpdesc; i++)
1486 m_zpdesc = get_next_zpdesc(m_zpdesc);
1487
1488 vaddr = kmap_local_zpdesc(m_zpdesc);
1489 link = (struct link_free *)vaddr + m_offset / sizeof(*link);
1490 set_freeobj(zspage, link->next >> OBJ_TAG_BITS);
1491 if (likely(!ZsHugePage(zspage)))
1492 /* record handle in the header of allocated chunk */
1493 link->handle = handle | OBJ_ALLOCATED_TAG;
1494 else
1495 zspage->first_zpdesc->handle = handle | OBJ_ALLOCATED_TAG;
1496
1497 kunmap_local(vaddr);
1498 mod_zspage_inuse(zspage, 1);
1499
1500 obj = location_to_obj(m_zpdesc, obj);
1501 record_obj(handle, obj);
1502
1503 return obj;
1504}

1521unsigned long zs_malloc(struct zs_pool *pool, size_t size, gfp_t gfp,
1522 const int nid)
1523{
1524 unsigned long handle;
1525 struct size_class *class;
1526 int newfg;
1527 struct zspage *zspage;
1528
1529 if (unlikely(!size))
1530 return (unsigned long)ERR_PTR(-EINVAL);
1531
1532 if (unlikely(size > ZS_MAX_ALLOC_SIZE))
1533 return (unsigned long)ERR_PTR(-ENOSPC);
1534
1535 handle = cache_alloc_handle(gfp);
1536 if (!handle)
1537 return (unsigned long)ERR_PTR(-ENOMEM);
1538
1539 /* extra space in chunk to keep the handle */
1540 size += ZS_HANDLE_SIZE;
1541 class = pool->size_class[get_size_class_index(size)];
1542
1543 /* class->lock effectively protects the zpage migration */
1544 spin_lock(&class->lock);
1545 zspage = find_get_zspage(class);
1546 if (likely(zspage)) {
1547 obj_malloc(pool, zspage, handle);
1548 /* Now move the zspage to another fullness group, if required */
1549 fix_fullness_group(class, zspage);
1550 class_stat_add(class, ZS_OBJS_INUSE, 1);
1551
1552 goto out;
1553 }
1554
1555 spin_unlock(&class->lock);
1556
1557 zspage = alloc_zspage(pool, class, gfp, nid);
1558 if (!zspage) {
1559 cache_free_handle(handle);
1560 return (unsigned long)ERR_PTR(-ENOMEM);
1561 }
1562
1563 spin_lock(&class->lock);
1564 obj_malloc(pool, zspage, handle);
1565 newfg = get_fullness_group(class, zspage);
1566 insert_zspage(class, zspage, newfg);
1567 atomic_long_add(class->pages_per_zspage, &pool->pages_allocated);
1568 class_stat_add(class, ZS_OBJS_ALLOCATED, class->objs_per_zspage);
1569 class_stat_add(class, ZS_OBJS_INUSE, 1);
1570
1571 /* We completely set up zspage so mark them as movable */
1572 SetZsPageMovable(pool, zspage);
1573out:
1574 spin_unlock(&class->lock);
1575
1576 return handle;
1577}

1652static void zs_object_copy(struct size_class *class, unsigned long dst,
1653 unsigned long src)
1654{
1655 struct zpdesc *s_zpdesc, *d_zpdesc;
1656 unsigned int s_objidx, d_objidx;
1657 unsigned long s_off, d_off;
1658 void *s_addr, *d_addr;
1659 int s_size, d_size, size;
1660 int written = 0;
1661
1662 s_size = d_size = class->size;
1663
1664 obj_to_location(src, &s_zpdesc, &s_objidx);
1665 obj_to_location(dst, &d_zpdesc, &d_objidx);
1666
1667 s_off = offset_in_page(class->size * s_objidx);
1668 d_off = offset_in_page(class->size * d_objidx);
1669
1670 if (s_off + class->size > PAGE_SIZE)
1671 s_size = PAGE_SIZE - s_off;
1672
1673 if (d_off + class->size > PAGE_SIZE)
1674 d_size = PAGE_SIZE - d_off;
1675
1676 s_addr = kmap_local_zpdesc(s_zpdesc);
1677 d_addr = kmap_local_zpdesc(d_zpdesc);
1678
1679 while (1) {
1680 size = min(s_size, d_size);
1681 memcpy(d_addr + d_off, s_addr + s_off, size);
1682 written += size;
1683
1684 if (written == class->size)
1685 break;
1686
1687 s_off += size;
1688 s_size -= size;
1689 d_off += size;
1690 d_size -= size;
1691
1692 /*
1693 * Calling kunmap_local(d_addr) is necessary. kunmap_local()
1694 * calls must occurs in reverse order of calls to kmap_local_page().
1695 * So, to call kunmap_local(s_addr) we should first call
1696 * kunmap_local(d_addr). For more details see
1697 * Documentation/mm/highmem.rst.
1698 */
1699 if (s_off >= PAGE_SIZE) {
1700 kunmap_local(d_addr);
1701 kunmap_local(s_addr);
1702 s_zpdesc = get_next_zpdesc(s_zpdesc);
1703 s_addr = kmap_local_zpdesc(s_zpdesc);
1704 d_addr = kmap_local_zpdesc(d_zpdesc);
1705 s_size = class->size - written;
1706 s_off = 0;
1707 }
1708
1709 if (d_off >= PAGE_SIZE) {
1710 kunmap_local(d_addr);
1711 d_zpdesc = get_next_zpdesc(d_zpdesc);
1712 d_addr = kmap_local_zpdesc(d_zpdesc);
1713 d_size = class->size - written;
1714 d_off = 0;
1715 }
1716 }
1717
1718 kunmap_local(d_addr);
1719 kunmap_local(s_addr);
1720}

1729static unsigned long find_alloced_obj(struct size_class *class,
1730 struct zpdesc *zpdesc, int *obj_idx)
1731{
1732 unsigned int offset;
1733 int index = *obj_idx;
1734 unsigned long handle = 0;
1735 void *addr = kmap_local_zpdesc(zpdesc);
1736
1737 offset = get_first_obj_offset(zpdesc);
1738 offset += class->size * index;

1742 break;
1743

1748 kunmap_local(addr);

1758static void migrate_zspage(struct zs_pool *pool, struct zspage *src_zspage,
1759 struct zspage *dst_zspage)
1760{
1761 unsigned long used_obj, free_obj;
1762 unsigned long handle;
1763 int obj_idx = 0;
1764 struct zpdesc *s_zpdesc = get_first_zpdesc(src_zspage);
1765 struct size_class *class = pool->size_class[src_zspage->class];
1766
1767 while (1) {
1768 handle = find_alloced_obj(class, s_zpdesc, &obj_idx);
1769 if (!handle) {
1770 s_zpdesc = get_next_zpdesc(s_zpdesc);
1771 if (!s_zpdesc)
1772 break;
1773 obj_idx = 0;
1774 continue;
1775 }
1776
1777 used_obj = handle_to_obj(handle);
1778 free_obj = obj_malloc(pool, dst_zspage, handle);
1779 zs_object_copy(class, free_obj, used_obj);
1780 obj_idx++;
1781 obj_free(class->size, used_obj);
1782
1783 /* Stop if there is no more space */
1784 if (zspage_full(class, dst_zspage))
1785 break;
1786
1787 /* Stop if there are no more objects to migrate */
1788 if (zspage_empty(src_zspage))
1789 break;
1790 }

1861static void lock_zspage(struct zspage *zspage)
1862{
1863 struct zpdesc *curr_zpdesc, *zpdesc;
1864
1865 /*
1866 * Pages we haven't locked yet can be migrated off the list while we're
1867 * trying to lock them, so we need to be careful and only attempt to
1868 * lock each page under zspage_read_lock(). Otherwise, the page we lock
1869 * may no longer belong to the zspage. This means that we may wait for
1870 * the wrong page to unlock, so we must take a reference to the page
1871 * prior to waiting for it to unlock outside zspage_read_lock().
1872 */
1873 while (1) {
1874 zspage_read_lock(zspage);
1875 zpdesc = get_first_zpdesc(zspage);
1876 if (zpdesc_trylock(zpdesc))
1877 break;
1878 zpdesc_get(zpdesc);
1879 zspage_read_unlock(zspage);
1880 zpdesc_wait_locked(zpdesc);
1881 zpdesc_put(zpdesc);
1882 }
1883
1884 curr_zpdesc = zpdesc;
1885 while ((zpdesc = get_next_zpdesc(curr_zpdesc))) {
1886 if (zpdesc_trylock(zpdesc)) {
1887 curr_zpdesc = zpdesc;
1888 } else {
1889 zpdesc_get(zpdesc);
1890 zspage_read_unlock(zspage);
1891 zpdesc_wait_locked(zpdesc);
1892 zpdesc_put(zpdesc);
1893 zspage_read_lock(zspage);
1894 }
1895 }

1945static int zs_page_migrate(struct page *newpage, struct page *page,
1946 enum migrate_mode mode)
1947{
1948 struct zs_pool *pool;
1949 struct size_class *class;
1950 struct zspage *zspage;
1951 struct zpdesc *dummy;
1952 struct zpdesc *newzpdesc = page_zpdesc(newpage);
1953 struct zpdesc *zpdesc = page_zpdesc(page);
1954 void *s_addr, *d_addr, *addr;
1955 unsigned int offset;
1956 unsigned long handle;
1957 unsigned long old_obj, new_obj;
1958 unsigned int obj_idx;
1959
1969 if (!zpdesc->zspage)
1970 return 0;
1971
1972 /* The page is locked, so this pointer must remain valid */
1973 zspage = get_zspage(zpdesc);
1974 pool = zspage->pool;
1975
1976 /*
1977 * The pool migrate_lock protects the race between zpage migration
1978 * and zs_free.
1979 */
1980 write_lock(&pool->lock);
1981 class = zspage_class(pool, zspage);
1982
1983 /*
1984 * the class lock protects zpage alloc/free in the zspage.
1985 */
1986 spin_lock(&class->lock);
1987 /* the zspage write_lock protects zpage access via zs_obj_read/write() */
1988 if (!zspage_write_trylock(zspage)) {

1992 }

1995 __zpdesc_set_zsmalloc(newzpdesc);
1996

1998 s_addr = kmap_local_zpdesc(zpdesc);
1999
2000 /*
2001 * Here, any user cannot access all objects in the zspage so let's move.
2002 */
2003 d_addr = kmap_local_zpdesc(newzpdesc);
2004 copy_page(d_addr, s_addr);
2005 kmsan_copy_page_meta(zpdesc_page(newzpdesc), zpdesc_page(zpdesc));
2006 kunmap_local(d_addr);
2007
2008 for (addr = s_addr + offset; addr < s_addr + PAGE_SIZE;
2009 addr += class->size) {
2010 if (obj_allocated(zpdesc, addr, &handle)) {
2011
2012 old_obj = handle_to_obj(handle);
2013 obj_to_location(old_obj, &dummy, &obj_idx);
2014 new_obj = (unsigned long)location_to_obj(newzpdesc, obj_idx);
2015 record_obj(handle, new_obj);
2016 }
2017 }
2018 kunmap_local(s_addr);
2019
2020 replace_sub_page(class, zspage, newzpdesc, zpdesc);
2021 /*
2022 * Since we complete the data copy and set up new zspage structure,
2023 * it's okay to release migration_lock.
2024 */
2025 write_unlock(&pool->lock);
2026 spin_unlock(&class->lock);
2027 zspage_write_unlock(zspage);
2028
2029 zpdesc_get(newzpdesc);
2030 if (zpdesc_zone(newzpdesc) != zpdesc_zone(zpdesc)) {
2031 zpdesc_dec_zone_page_state(zpdesc);
2032 zpdesc_inc_zone_page_state(newzpdesc);
2033 }
2034
2035 reset_zpdesc(zpdesc);

2159static unsigned long __zs_compact(struct zs_pool *pool,
2160 struct size_class *class)
2161{
2162 struct zspage *src_zspage = NULL;
2163 struct zspage *dst_zspage = NULL;
2164 unsigned long pages_freed = 0;
2165
2166 /*
2167 * protect the race between zpage migration and zs_free
2168 * as well as zpage allocation/free
2169 */
2170 write_lock(&pool->lock);
2171 spin_lock(&class->lock);
2172 while (zs_can_compact(class)) {
2173 int fg;
2174
2175 if (!dst_zspage) {
2176 dst_zspage = isolate_dst_zspage(class);
2177 if (!dst_zspage)
2178 break;
2179 }
2180
2181 src_zspage = isolate_src_zspage(class);
2182 if (!src_zspage)
2183 break;
2184
2185 if (!zspage_write_trylock(src_zspage))
2186 break;
2187
2188 migrate_zspage(pool, src_zspage, dst_zspage);
2189 zspage_write_unlock(src_zspage);
2190
2191 fg = putback_zspage(class, src_zspage);
2192 if (fg == ZS_INUSE_RATIO_0) {
2193 free_zspage(pool, class, src_zspage);
2194 pages_freed += class->pages_per_zspage;
2195 }
2196 src_zspage = NULL;
2197
2198 if (get_fullness_group(class, dst_zspage) == ZS_INUSE_RATIO_100
2199 || rwlock_is_contended(&pool->lock)) {
2200 putback_zspage(class, dst_zspage);
2201 dst_zspage = NULL;
2202
2203 spin_unlock(&class->lock);
2204 write_unlock(&pool->lock);
2205 cond_resched();
2206 write_lock(&pool->lock);
2207 spin_lock(&class->lock);
2208 }
2209 }
2210
2211 if (src_zspage)
2212 putback_zspage(class, src_zspage);
2213
2214 if (dst_zspage)
2215 putback_zspage(class, dst_zspage);
2216
2217 spin_unlock(&class->lock);
2218 write_unlock(&pool->lock);
2219
2220 return pages_freed;
2221}

2226unsigned long zs_compact(struct zs_pool *pool)
2227{
2228 int i;
2229 struct size_class *class;
2230 unsigned long pages_freed = 0;
2231
2232 /*
2233 * Pool compaction is performed under pool->lock so it is basically
2234 * single-threaded. Having more than one thread in __zs_compact()
2235 * will increase pool->lock contention, which will impact other
2236 * zsmalloc operations that need pool->lock.
2237 */
2238 if (atomic_xchg(&pool->compaction_in_progress, 1))
2239 return 0;
2240

2242 class = pool->size_class[i];
2243 if (class->index != i)
2244 continue;
2245 pages_freed += __zs_compact(pool, class);
2246 }
2247 atomic_long_add(pages_freed, &pool->stats.pages_compacted);
2248 atomic_set(&pool->compaction_in_progress, 0);
2249
2250 return pages_freed;
2251}

2368struct zs_pool *zs_create_pool(const char *name)
2369{
2370 int i;
2371 struct zs_pool *pool;
2372 struct size_class *prev_class = NULL;
2373
2374 pool = kzalloc_obj(*pool);
2375 if (!pool)
2376 return NULL;
2377
2378 init_deferred_free(pool);
2379 rwlock_init(&pool->lock);
2380 atomic_set(&pool->compaction_in_progress, 0);
2381
2382 pool->name = kstrdup(name, GFP_KERNEL);
2383 if (!pool->name)
2384 goto err;
2385
2386 /*
2387 * Iterate reversely, because, size of size_class that we want to use
2388 * for merging should be larger or equal to current size.
2389 */
2390 for (i = ZS_SIZE_CLASSES - 1; i >= 0; i--) {
2391 int size;
2392 int pages_per_zspage;
2393 int objs_per_zspage;
2394 struct size_class *class;
2395 int fullness;
2396
2397 size = ZS_MIN_ALLOC_SIZE + i * ZS_SIZE_CLASS_DELTA;
2398 if (size > ZS_MAX_ALLOC_SIZE)
2399 size = ZS_MAX_ALLOC_SIZE;
2400 pages_per_zspage = calculate_zspage_chain_size(size);
2401 objs_per_zspage = pages_per_zspage * PAGE_SIZE / size;
2402
2403 /*
2404 * We iterate from biggest down to smallest classes,
2405 * so huge_class_size holds the size of the first huge
2406 * class. Any object bigger than or equal to that will
2407 * endup in the huge class.
2408 */
2409 if (pages_per_zspage != 1 && objs_per_zspage != 1 &&
2410 !huge_class_size) {
2411 huge_class_size = size;
2412 /*
2413 * The object uses ZS_HANDLE_SIZE bytes to store the
2414 * handle. We need to subtract it, because zs_malloc()
2415 * unconditionally adds handle size before it performs
2416 * size class search - so object may be smaller than
2417 * huge class size, yet it still can end up in the huge
2418 * class because it grows by ZS_HANDLE_SIZE extra bytes
2419 * right before class lookup.
2420 */
2421 huge_class_size -= (ZS_HANDLE_SIZE - 1);
2422 }
2423
2424 /*
2425 * size_class is used for normal zsmalloc operation such
2426 * as alloc/free for that size. Although it is natural that we
2427 * have one size_class for each size, there is a chance that we
2428 * can get more memory utilization if we use one size_class for
2429 * many different sizes whose size_class have same
2430 * characteristics. So, we makes size_class point to
2431 * previous size_class if possible.
2432 */
2433 if (prev_class) {
2434 if (can_merge(prev_class, pages_per_zspage, objs_per_zspage)) {
2435 pool->size_class[i] = prev_class;
2436 continue;
2437 }
2438 }
2439
2440 class = kzalloc_obj(struct size_class);
2441 if (!class)
2442 goto err;
2443
2444 class->size = size;
2445 class->index = i;
2446 class->pages_per_zspage = pages_per_zspage;
2447 class->objs_per_zspage = objs_per_zspage;
2448 spin_lock_init(&class->lock);
2449 pool->size_class[i] = class;
2450
2451 fullness = ZS_INUSE_RATIO_0;
2452 while (fullness < NR_FULLNESS_GROUPS) {
2453 INIT_LIST_HEAD(&class->fullness_list[fullness]);
2454 fullness++;
2455 }
2456
2457 prev_class = class;
2458 }
2459
2460 /* debug only, don't abort if it fails */
2461 zs_pool_stat_create(pool, name);
2462
2463 /*
2464 * Not critical since shrinker is only used to trigger internal
2465 * defragmentation of the pool which is pretty optional thing. If
2466 * registration fails we still can use the pool normally and user can
2467 * trigger compaction manually. Thus, ignore return code.
2468 */
2469 zs_register_shrinker(pool);
2470
2471 return pool;
2472
2473err:
2474 zs_destroy_pool(pool);
2475 return NULL;
2476}
