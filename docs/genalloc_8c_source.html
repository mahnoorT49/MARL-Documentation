2/*
3 * Basic general purpose allocator for managing special purpose
4 * memory, for example, memory that is not managed by the regular
5 * kmalloc/kfree interface. Uses for this includes on-device special
6 * memory, uncached memory etc.
7 *
8 * It is safe to use the allocator in NMI handlers and other special
9 * unblockable contexts that could otherwise deadlock on locks. This
10 * is implemented by using atomic operations and retries on any
11 * conflicts. The disadvantage is that there may be livelocks in
12 * extreme cases. For better scalability, one allocator can be used
13 * for each CPU.
14 *
15 * The lockless operation only works if there is enough memory
16 * available. If new memory is added to the pool a lock has to be
17 * still taken. So any user relying on locklessness has to ensure
18 * that sufficient memory is preallocated.
19 *
20 * The basic atomic operation of this allocator is cmpxchg on long.
21 * On architectures that don't have NMI-safe cmpxchg implementation,
22 * the allocator can NOT be used in NMI handler. So code uses the
23 * allocator in NMI handler should depend on
24 * CONFIG_ARCH_HAVE_NMI_SAFE_CMPXCHG.
25 *
26 * Copyright 2005 (C) Jes Sorensen <jes@trained-monkey.org>
27 */

51static inline int
52set_bits_ll(unsigned long *addr, unsigned long mask_to_set)
53{
54 unsigned long val = READ_ONCE(*addr);
55
56 do {
57 if (val & mask_to_set)
58 return -EBUSY;
59 cpu_relax();
60 } while (!try_cmpxchg(addr, &val, val | mask_to_set));
61
62 return 0;
63}
64
68static inline int
69clear_bits_ll(unsigned long *addr, unsigned long mask_to_clear)
70{
71 unsigned long val = READ_ONCE(*addr);
72
73 do {
74 if ((val & mask_to_clear) != mask_to_clear)
75 return -EBUSY;
76 cpu_relax();
77 } while (!try_cmpxchg(addr, &val, val & ~mask_to_clear));
78
79 return 0;
80}

96static unsigned long
97bitmap_set_ll(unsigned long *map, unsigned long start, unsigned long nr)
98{
99 unsigned long *p = map + BIT_WORD(start);
100 const unsigned long size = start + nr;
101 int bits_to_set = BITS_PER_LONG - (start % BITS_PER_LONG);
102 unsigned long mask_to_set = BITMAP_FIRST_WORD_MASK(start);
103
104 while (nr >= bits_to_set) {
105 if (set_bits_ll(p, mask_to_set))
106 return nr;
107 nr -= bits_to_set;
108 bits_to_set = BITS_PER_LONG;
109 mask_to_set = ~0UL;
110 p++;
111 }
112 if (nr) {
113 mask_to_set &= BITMAP_LAST_WORD_MASK(size);
114 if (set_bits_ll(p, mask_to_set))
115 return nr;
116 }
117
118 return 0;
119}

135static unsigned long
136bitmap_clear_ll(unsigned long *map, unsigned long start, unsigned long nr)
137{
138 unsigned long *p = map + BIT_WORD(start);
139 const unsigned long size = start + nr;
140 int bits_to_clear = BITS_PER_LONG - (start % BITS_PER_LONG);
141 unsigned long mask_to_clear = BITMAP_FIRST_WORD_MASK(start);
142
143 while (nr >= bits_to_clear) {
144 if (clear_bits_ll(p, mask_to_clear))
145 return nr;
146 nr -= bits_to_clear;
147 bits_to_clear = BITS_PER_LONG;
148 mask_to_clear = ~0UL;
149 p++;
150 }
151 if (nr) {
152 mask_to_clear &= BITMAP_LAST_WORD_MASK(size);
153 if (clear_bits_ll(p, mask_to_clear))
154 return nr;
155 }
156
157 return 0;
158}

171struct gen_pool *gen_pool_create(int min_alloc_order, int nid)
172{
173 struct gen_pool *pool;
174
175 pool = kmalloc_node(sizeof(struct gen_pool), GFP_KERNEL, nid);
176 if (pool != NULL) {
177 spin_lock_init(&pool->lock);
178 INIT_LIST_HEAD(&pool->chunks);
179 pool->min_alloc_order = min_alloc_order;
180 pool->algo = gen_pool_first_fit;
181 pool->data = NULL;
182 pool->name = NULL;
183 }
184 return pool;
185}

205int gen_pool_add_owner(struct gen_pool *pool, unsigned long virt, phys_addr_t phys,
206 size_t size, int nid, void *owner)
207{
208 struct gen_pool_chunk *chunk;
209 unsigned long nbits = size >> pool->min_alloc_order;
210 unsigned long nbytes = sizeof(struct gen_pool_chunk) +
211 BITS_TO_LONGS(nbits) * sizeof(long);
212
213 chunk = vzalloc_node(nbytes, nid);
214 if (unlikely(chunk == NULL))
215 return -ENOMEM;
216
217 chunk->phys_addr = phys;
218 chunk->start_addr = virt;
219 chunk->end_addr = virt + size - 1;
220 chunk->owner = owner;
221 atomic_long_set(&chunk->avail, size);
222
223 spin_lock(&pool->lock);
224 list_add_rcu(&chunk->next_chunk, &pool->chunks);
225 spin_unlock(&pool->lock);
226
227 return 0;
228}
229EXPORT_SYMBOL(gen_pool_add_owner);

241phys_addr_t gen_pool_virt_to_phys(struct gen_pool *pool, unsigned long addr)
242{
243 struct gen_pool_chunk *chunk;
244 phys_addr_t paddr = -1;
245
246 rcu_read_lock();
247 list_for_each_entry_rcu(chunk, &pool->chunks, next_chunk) {
248 if (addr >= chunk->start_addr && addr <= chunk->end_addr) {
249 paddr = chunk->phys_addr + (addr - chunk->start_addr);
250 break;
251 }
252 }
253 rcu_read_unlock();
254
255 return paddr;
256}

269void gen_pool_destroy(struct gen_pool *pool)
270{
271 struct list_head *_chunk, *_next_chunk;
272 struct gen_pool_chunk *chunk;
273 int order = pool->min_alloc_order;
274 unsigned long bit, end_bit;
275
276 list_for_each_safe(_chunk, _next_chunk, &pool->chunks) {
277 chunk = list_entry(_chunk, struct gen_pool_chunk, next_chunk);
278 list_del(&chunk->next_chunk);
279
280 end_bit = chunk_size(chunk) >> order;
281 bit = find_first_bit(chunk->bits, end_bit);
282 BUG_ON(bit < end_bit);
283
284 vfree(chunk);
285 }
286 kfree_const(pool->name);
287 kfree(pool);
288}
289EXPORT_SYMBOL(gen_pool_destroy);

307unsigned long gen_pool_alloc_algo_owner(struct gen_pool *pool, size_t size,
308 genpool_algo_t algo, void *data, void **owner)
309{
310 struct gen_pool_chunk *chunk;
311 unsigned long addr = 0;
312 int order = pool->min_alloc_order;
313 unsigned long nbits, start_bit, end_bit, remain;
314
315#ifndef CONFIG_ARCH_HAVE_NMI_SAFE_CMPXCHG
316 BUG_ON(in_nmi());
317#endif
318
319 if (owner)
320 *owner = NULL;
321
322 if (size == 0)
323 return 0;
324
325 nbits = (size + (1UL << order) - 1) >> order;
326 rcu_read_lock();
327 list_for_each_entry_rcu(chunk, &pool->chunks, next_chunk) {
328 if (size > atomic_long_read(&chunk->avail))
329 continue;
330
331 start_bit = 0;
332 end_bit = chunk_size(chunk) >> order;
333retry:
334 start_bit = algo(chunk->bits, end_bit, start_bit,
335 nbits, data, pool, chunk->start_addr);
336 if (start_bit >= end_bit)
337 continue;
338 remain = bitmap_set_ll(chunk->bits, start_bit, nbits);
339 if (remain) {
340 remain = bitmap_clear_ll(chunk->bits, start_bit,
341 nbits - remain);
342 BUG_ON(remain);
343 goto retry;
344 }
345
346 addr = chunk->start_addr + ((unsigned long)start_bit << order);
347 size = nbits << order;
348 atomic_long_sub(size, &chunk->avail);
349 if (owner)
350 *owner = chunk->owner;
351 break;
352 }
353 rcu_read_unlock();
354 return addr;
355}
356EXPORT_SYMBOL(gen_pool_alloc_algo_owner);

538void gen_pool_free_owner(struct gen_pool *pool, unsigned long addr, size_t size,
539 void **owner)
540{
541 struct gen_pool_chunk *chunk;
542 int order = pool->min_alloc_order;
543 unsigned long start_bit, nbits, remain;
544
545#ifndef CONFIG_ARCH_HAVE_NMI_SAFE_CMPXCHG
546 BUG_ON(in_nmi());
547#endif
548
549 if (owner)
550 *owner = NULL;
551
552 nbits = (size + (1UL << order) - 1) >> order;
553 rcu_read_lock();
554 list_for_each_entry_rcu(chunk, &pool->chunks, next_chunk) {
555 if (addr >= chunk->start_addr && addr <= chunk->end_addr) {
556 BUG_ON(addr + size - 1 > chunk->end_addr);
557 start_bit = (addr - chunk->start_addr) >> order;
558 remain = bitmap_clear_ll(chunk->bits, start_bit, nbits);
559 BUG_ON(remain);
560 size = nbits << order;
561 atomic_long_add(size, &chunk->avail);
562 if (owner)
563 *owner = chunk->owner;
564 rcu_read_unlock();
565 return;
566 }
567 }
568 rcu_read_unlock();
569 BUG();
570}
571EXPORT_SYMBOL(gen_pool_free_owner);

610bool gen_pool_has_addr(struct gen_pool *pool, unsigned long start,
611 size_t size)
612{
613 bool found = false;

615 struct gen_pool_chunk *chunk;
616
617 rcu_read_lock();
618 list_for_each_entry_rcu(chunk, &(pool)->chunks, next_chunk) {
619 if (start >= chunk->start_addr && start <= chunk->end_addr) {
620 if (end <= chunk->end_addr) {
621 found = true;
622 break;
623 }
624 }

626 rcu_read_unlock();

768unsigned long gen_pool_fixed_alloc(unsigned long *map, unsigned long size,
769 unsigned long start, unsigned int nr, void *data,
770 struct gen_pool *pool, unsigned long start_addr)
771{
772 struct genpool_data_fixed *fixed_data;
773 int order;
774 unsigned long offset_bit;
775 unsigned long start_bit;
776
777 fixed_data = data;
778 order = pool->min_alloc_order;
779 offset_bit = fixed_data->offset >> order;
780 if (WARN_ON(fixed_data->offset & ((1UL << order) - 1)))
781 return size;
782
783 start_bit = bitmap_find_next_zero_area(map, size,
784 start + offset_bit, nr, 0);
785 if (start_bit != offset_bit)
786 start_bit = size;
787 return start_bit;
788}
789EXPORT_SYMBOL(gen_pool_fixed_alloc);

834unsigned long gen_pool_best_fit(unsigned long *map, unsigned long size,
835 unsigned long start, unsigned int nr, void *data,
836 struct gen_pool *pool, unsigned long start_addr)
837{
838 unsigned long start_bit = size;
839 unsigned long len = size + 1;
840 unsigned long index;
841
842 index = bitmap_find_next_zero_area(map, size, start, nr, 0);
843
844 while (index < size) {
845 unsigned long next_bit = find_next_bit(map, size, index + nr);
846 if ((next_bit - index) < len) {
847 len = next_bit - index;
848 start_bit = index;
849 if (len == nr)
850 return start_bit;
851 }
852 index = bitmap_find_next_zero_area(map, size,
853 next_bit + 1, nr, 0);
854 }
855
856 return start_bit;
857}
858EXPORT_SYMBOL(gen_pool_best_fit);

868/*
869 * Function Description: This is a match function used to find a specific pool during device management operations. It compares the pool's name with the provided name to determine if the pool matches the search criteria.
870*/
871static int devm_gen_pool_match(struct device *dev, void *res, void *data)
872{
873 struct gen_pool **p = res;
874
875 /* NULL data matches only a pool without an assigned name */
876 if (!data && !(*p)->name)
877 return 1;
878
879 if (!data || !(*p)->name)
880 return 0;
881
882 return !strcmp((*p)->name, data);
883}

921struct gen_pool *devm_gen_pool_create(struct device *dev, int min_alloc_order,
922 int nid, const char *name)
923{
924 struct gen_pool **ptr, *pool;
925 const char *pool_name = NULL;
926
927 /* Check that genpool to be created is uniquely addressed on device */
928 if (gen_pool_get(dev, name))
929 return ERR_PTR(-EINVAL);
930
931 if (name) {
932 pool_name = kstrdup_const(name, GFP_KERNEL);
933 if (!pool_name)
934 return ERR_PTR(-ENOMEM);
935 }
936
937 ptr = devres_alloc(devm_gen_pool_release, sizeof(*ptr), GFP_KERNEL);
938 if (!ptr)
939 goto free_pool_name;
940
941 pool = gen_pool_create(min_alloc_order, nid);
942 if (!pool)
943 goto free_devres;
944
945 *ptr = pool;
946 pool->name = pool_name;
947 devres_add(dev, ptr);
948
949 return pool;
950
951free_devres:
952 devres_free(ptr);
953free_pool_name:
954 kfree_const(pool_name);
955
956 return ERR_PTR(-ENOMEM);
957}
