18/*
19 * Determines whether a given css can participate in rstat.
20 * css's that are cgroup::self use rstat for base stats.
21 * Other css's associated with a subsystem use rstat only when
22 * they define the ss->css_rstat_flush callback.
23 *
24 *
25 * Description: Checks whether a given cgroup subsystem state (css) can participate in rstat. It returns true if the css is the cgroup's self css or if the subsystem has defined a flush callback. This determines if rstat tracking should be enabled for this css.
26 */
27static inline bool css_uses_rstat(struct cgroup_subsys_state *css)
28{
29 return css_is_self(css) || css->ss->css_rstat_flush != NULL;
30}

89void __css_rstat_updated(struct cgroup_subsys_state *css, int cpu)
90{
91 struct llist_head *lhead;
92 struct css_rstat_cpu *rstatc;
93 struct llist_node *self;
94
95 /* Prevent access to uninitialized rstat pointers. */
96 if (!css_uses_rstat(css))
97 return;
98
99 lockdep_assert_preemption_disabled();
100
101 /*
102 * The lockless insertion below relies on NMI-safe cmpxchg;
103 * bail out in NMI on archs that don't provide it.
104 */
105 if (!IS_ENABLED(CONFIG_ARCH_HAVE_NMI_SAFE_CMPXCHG) && in_nmi())
106 return;
107
108 rstatc = css_rstat_cpu(css, cpu);
109 /*
110 * If already on list return. This check is racy and smp_mb() is needed
111 * to pair it with the smp_mb() in css_process_update_tree() if the
112 * guarantee that the updated stats are visible to concurrent flusher is
113 * needed.
114 */
115 if (llist_on_list(&rstatc->lnode))
116 return;
117
118 /*
119 * This function can be renentered by irqs and nmis for the same cgroup
120 * and may try to insert the same per-cpu lnode into the llist. Note
121 * that llist_add() does not protect against such scenarios. In addition
122 * this same per-cpu lnode can be modified through init_llist_node()
123 * from css_rstat_flush() running on a different CPU.
124 *
125 * To protect against such stacked contexts of irqs/nmis, we use the
126 * fact that lnode points to itself when not on a list and then use
127 * try_cmpxchg() to atomically set to NULL to select the winner
128 * which will call llist_add(). The losers can assume the insertion is
129 * successful and the winner will eventually add the per-cpu lnode to
130 * the llist.
131 *
132 * Please note that we can not use this_cpu_cmpxchg() here as on some
133 * archs it is not safe against modifications from multiple CPUs.
134 */
135 self = &rstatc->lnode;
136 if (!try_cmpxchg(&rstatc->lnode.next, &self, NULL))
137 return;
138
139 lhead = ss_lhead_cpu(css->ss, cpu);
140 llist_add(&rstatc->lnode, lhead);
141}

161static void __css_process_update_tree(struct cgroup_subsys_state *css, int cpu)
162{
163 /* put @css and all ancestors on the corresponding updated lists */
164 while (true) {
165 struct css_rstat_cpu *rstatc = css_rstat_cpu(css, cpu);
166 struct cgroup_subsys_state *parent = css->parent;
167 struct css_rstat_cpu *prstatc;
168
169 /*
170 * Both additions and removals are bottom-up. If a cgroup
171 * is already in the tree, all ancestors are.
172 */
173 if (rstatc->updated_next)
174 break;
175
176 /* Root has no parent to link it to, but mark it busy */
177 if (!parent) {
178 rstatc->updated_next = css;
179 break;
180 }
181
182 prstatc = css_rstat_cpu(parent, cpu);
183 rstatc->updated_next = prstatc->updated_children;
184 prstatc->updated_children = css;
185
186 css = parent;
187 }
188}

193static void css_process_update_tree(struct cgroup_subsys *ss, int cpu)
194{
195 struct llist_head *lhead = ss_lhead_cpu(ss, cpu);
196 struct llist_node *lnode;
197
198 while ((lnode = llist_del_first_init(lhead))) {
199 struct css_rstat_cpu *rstatc;
200
201 /*
202 * smp_mb() is needed here (more specifically in between
203 * init_llist_node() and per-cpu stats flushing) if the
204 * guarantee is required by a rstat user where etiher the
205 * updater should add itself on the lockless list or the
206 * flusher flush the stats updated by the updater who have
207 * observed that they are already on the list. The
208 * corresponding barrier pair for this one should be before
209 * __css_rstat_updated() by the user.
210 *
211 * For now, there aren't any such user, so not adding the
212 * barrier here but if such a use-case arise, please add
213 * smp_mb() here.
214 */
215
216 rstatc = container_of(lnode, struct css_rstat_cpu, lnode);
217 __css_process_update_tree(rstatc->owner, cpu);
218 }
219}

237static struct cgroup_subsys_state *css_rstat_push_children(
238 struct cgroup_subsys_state *head,
239 struct cgroup_subsys_state *child, int cpu)
240{
241 struct cgroup_subsys_state *cnext = child; /* Next head of child css level */
242 struct cgroup_subsys_state *ghead = NULL; /* Head of grandchild css level */
243 struct cgroup_subsys_state *parent, *grandchild;
244 struct css_rstat_cpu *crstatc;
245
246 child->rstat_flush_next = NULL;
247
248 /*
249 * The subsystem rstat lock must be held for the whole duration from
250 * here as the rstat_flush_next list is being constructed to when
251 * it is consumed later in css_rstat_flush().
252 */
253 lockdep_assert_held(ss_rstat_lock(head->ss));
254
255 /*
256 * Notation: -> updated_next pointer
257 * => rstat_flush_next pointer
258 *
259 * Assuming the following sample updated_children lists:
260 * P: C1 -> C2 -> P
261 * C1: G11 -> G12 -> C1
262 * C2: G21 -> G22 -> C2
263 *
264 * After 1st iteration:
265 * head => C2 => C1 => NULL
266 * ghead => G21 => G11 => NULL
267 *
268 * After 2nd iteration:
269 * head => G12 => G11 => G22 => G21 => C2 => C1 => NULL
270 */
271next_level:
272 while (cnext) {
273 child = cnext;
274 cnext = child->rstat_flush_next;
275 parent = child->parent;
276
277 /* updated_next is parent cgroup terminated if !NULL */
278 while (child != parent) {
279 child->rstat_flush_next = head;
280 head = child;
281 crstatc = css_rstat_cpu(child, cpu);
282 grandchild = crstatc->updated_children;
283 if (grandchild != child) {
284 /* Push the grand child to the next level */
285 crstatc->updated_children = child;
286 grandchild->rstat_flush_next = ghead;
287 ghead = grandchild;
288 }
289 child = crstatc->updated_next;
290 crstatc->updated_next = NULL;
291 }
292 }
293
294 if (ghead) {
295 cnext = ghead;
296 ghead = NULL;
297 goto next_level;
298 }
299 return head;
300}

323static struct cgroup_subsys_state *css_rstat_updated_list(
324 struct cgroup_subsys_state *root, int cpu)
325{
326 struct css_rstat_cpu *rstatc = css_rstat_cpu(root, cpu);
327 struct cgroup_subsys_state *head = NULL, *parent, *child;
328
329 css_process_update_tree(root->ss, cpu);
330
331 /* Return NULL if this subtree is not on-list */
332 if (!rstatc->updated_next)
333 return NULL;
334
335 /*
336 * Unlink @root from its parent. As the updated_children list is
337 * singly linked, we have to walk it to find the removal point.
338 */
339 parent = root->parent;
340 if (parent) {
341 struct css_rstat_cpu *prstatc;
342 struct cgroup_subsys_state **nextp;
343
344 prstatc = css_rstat_cpu(parent, cpu);
345 nextp = &prstatc->updated_children;
346 while (*nextp != root) {
347 struct css_rstat_cpu *nrstatc;
348
349 nrstatc = css_rstat_cpu(*nextp, cpu);
350 WARN_ON_ONCE(*nextp == parent);
351 nextp = &nrstatc->updated_next;
352 }
353 *nextp = rstatc->updated_next;
354 }
355
356 rstatc->updated_next = NULL;
357
358 /* Push @root to the list first before pushing the children */
359 head = root;
360 root->rstat_flush_next = NULL;
361 child = rstatc->updated_children;
362 rstatc->updated_children = root;
363 if (child != root)
364 head = css_rstat_push_children(head, child, cpu);
365
366 return head;
367}

394/*
395 * Helper functions for locking.
396 *
397 * This makes it easier to diagnose locking issues and contention in
398 * production environments. The parameter @cpu_in_loop indicate lock
399 * was released and re-taken when collection data from the CPUs. The
400 * value -1 is used when obtaining the main lock else this is the CPU
401 * number processed last.
402 *
403 *
404 * Function Description: Acquires the rstat lock for a given css with tracing and contention detection. It tries to acquire the lock without blocking first, and if contention is detected, it traces the event and then waits for the lock. This helps diagnose locking issues in production.
405 */
406static inline void __css_rstat_lock(struct cgroup_subsys_state *css,
407 int cpu_in_loop)
408 __acquires(ss_rstat_lock(css->ss))
409{
410 struct cgroup *cgrp = css->cgroup;
411 spinlock_t *lock;
412 bool contended;
413
414 lock = ss_rstat_lock(css->ss);
415 contended = !spin_trylock_irq(lock);
416 if (contended) {
417 trace_cgroup_rstat_lock_contended(cgrp, cpu_in_loop, contended);
418 spin_lock_irq(lock);
419 }
420 trace_cgroup_rstat_locked(cgrp, cpu_in_loop, contended);
421}
422
426static inline void __css_rstat_unlock(struct cgroup_subsys_state *css,
427 int cpu_in_loop)
428 __releases(ss_rstat_lock(css->ss))
429{
430 struct cgroup *cgrp = css->cgroup;
431 spinlock_t *lock;
432
433 lock = ss_rstat_lock(css->ss);
434 trace_cgroup_rstat_unlock(cgrp, cpu_in_loop, false);
435 spin_unlock_irq(lock);
436}

454__bpf_kfunc void css_rstat_flush(struct cgroup_subsys_state *css)
455{
456 int cpu;
457 bool is_self = css_is_self(css);
458
459 /*
460 * Since bpf programs can call this function, prevent access to
461 * uninitialized rstat pointers.
462 */
463 if (!css_uses_rstat(css))
464 return;
465
466 might_sleep();
467 for_each_possible_cpu(cpu) {
468 struct cgroup_subsys_state *pos;
469
470 /* Reacquire for each CPU to avoid disabling IRQs too long */
471 __css_rstat_lock(css, cpu);
472 pos = css_rstat_updated_list(css, cpu);
473 for (; pos; pos = pos->rstat_flush_next) {
474 if (is_self) {
475 cgroup_base_stat_flush(pos->cgroup, cpu);
476 bpf_rstat_flush(pos->cgroup,
477 cgroup_parent(pos->cgroup), cpu);
478 } else
479 pos->ss->css_rstat_flush(pos, cpu);
480 }
481 __css_rstat_unlock(css, cpu);
482 if (!cond_resched())
483 cpu_relax();
484 }
485}

490int css_rstat_init(struct cgroup_subsys_state *css)
491{
492 struct cgroup *cgrp = css->cgroup;
493 int cpu;
494 bool is_self = css_is_self(css);
495
496 if (is_self) {
497 /* the root cgrp has rstat_base_cpu preallocated */
498 if (!cgrp->rstat_base_cpu) {
499 cgrp->rstat_base_cpu = alloc_percpu(struct cgroup_rstat_base_cpu);
500 if (!cgrp->rstat_base_cpu)
501 return -ENOMEM;
502 }
503 } else if (css->ss->css_rstat_flush == NULL)
504 return 0;
505
506 /* the root cgrp's self css has rstat_cpu preallocated */
507 if (!css->rstat_cpu) {
508 css->rstat_cpu = alloc_percpu(struct css_rstat_cpu);
509 if (!css->rstat_cpu) {
510 if (is_self)
511 free_percpu(cgrp->rstat_base_cpu);
512
513 return -ENOMEM;
514 }
515 }
516
517 /* ->updated_children list is self terminated */
518 for_each_possible_cpu(cpu) {
519 struct css_rstat_cpu *rstatc = css_rstat_cpu(css, cpu);
520
521 rstatc->owner = rstatc->updated_children = css;
522 init_llist_node(&rstatc->lnode);
523
524 if (is_self) {
525 struct cgroup_rstat_base_cpu *rstatbc;
526
527 rstatbc = cgroup_rstat_base_cpu(cgrp, cpu);
528 u64_stats_init(&rstatbc->bsync);
529 }
530 }
531
532 return 0;
533}

538void css_rstat_exit(struct cgroup_subsys_state *css)
539{
540 int cpu;
541
542 if (!css_uses_rstat(css))
543 return;
544
545 if (!css->rstat_cpu)
546 return;
547
548 css_rstat_flush(css);
549
550 /* sanity check */
551 for_each_possible_cpu(cpu) {
552 struct css_rstat_cpu *rstatc = css_rstat_cpu(css, cpu);
553
554 if (WARN_ON_ONCE(rstatc->updated_children != css) ||
555 WARN_ON_ONCE(rstatc->updated_next))
556 return;
557 }
558
559 if (css_is_self(css)) {
560 struct cgroup *cgrp = css->cgroup;
561
562 free_percpu(cgrp->rstat_base_cpu);
563 cgrp->rstat_base_cpu = NULL;
564 }
565
566 free_percpu(css->rstat_cpu);
567 css->rstat_cpu = NULL;
568}

635static void cgroup_base_stat_flush(struct cgroup *cgrp, int cpu)
636{
637 struct cgroup_rstat_base_cpu *rstatbc = cgroup_rstat_base_cpu(cgrp, cpu);
638 struct cgroup *parent = cgroup_parent(cgrp);
639 struct cgroup_rstat_base_cpu *prstatbc;
640 struct cgroup_base_stat delta;
641 unsigned seq;
642
643 /* Root-level stats are sourced from system-wide CPU stats */
644 if (!parent)
645 return;
646
647 /* fetch the current per-cpu values */
648 do {
649 seq = __u64_stats_fetch_begin(&rstatbc->bsync);
650 delta = rstatbc->bstat;
651 } while (__u64_stats_fetch_retry(&rstatbc->bsync, seq));
652
653 /* propagate per-cpu delta to cgroup and per-cpu global statistics */
654 cgroup_base_stat_sub(&delta, &rstatbc->last_bstat);
655 cgroup_base_stat_add(&cgrp->bstat, &delta);
656 cgroup_base_stat_add(&rstatbc->last_bstat, &delta);
657 cgroup_base_stat_add(&rstatbc->subtree_bstat, &delta);
658
659 /* propagate cgroup and per-cpu global delta to parent (unless that's root) */
660 if (cgroup_parent(parent)) {
661 delta = cgrp->bstat;
662 cgroup_base_stat_sub(&delta, &cgrp->last_bstat);
663 cgroup_base_stat_add(&parent->bstat, &delta);
664 cgroup_base_stat_add(&cgrp->last_bstat, &delta);
665
666 delta = rstatbc->subtree_bstat;
667 prstatbc = cgroup_rstat_base_cpu(parent, cpu);
668 cgroup_base_stat_sub(&delta, &rstatbc->last_subtree_bstat);
669 cgroup_base_stat_add(&prstatbc->subtree_bstat, &delta);
670 cgroup_base_stat_add(&rstatbc->last_subtree_bstat, &delta);
671 }
672}

715void __cgroup_account_cputime_field(struct cgroup *cgrp,
716 enum cpu_usage_stat index, u64 delta_exec)
717{
718 struct cgroup_rstat_base_cpu *rstatbc;
719 unsigned long flags;
720
721 rstatbc = cgroup_base_stat_cputime_account_begin(cgrp, &flags);

727 case CPUTIME_USER:

729 break;
730 case CPUTIME_SYSTEM:
731 case CPUTIME_IRQ:
732 case CPUTIME_SOFTIRQ:
733 rstatbc->bstat.cputime.stime += delta_exec;
734 break;
735#ifdef CONFIG_SCHED_CORE
736 case CPUTIME_FORCEIDLE:
737 rstatbc->bstat.forceidle_sum += delta_exec;
738 break;
739#endif
740 default:
741 break;
742 }
743
744 cgroup_base_stat_cputime_account_end(cgrp, rstatbc, flags);
745}

747/*
748 * compute the cputime for the root cgroup by getting the per cpu data
749 * at a global level, then categorizing the fields in a manner consistent
750 * with how it is done by __cgroup_account_cputime_field for each bit of
751 * cpu time attributed to a cgroup.
752 *
753 *
754 * Function Description: Computes the CPU time statistics for the root cgroup by aggregating per-CPU kernel cpustat data. It reads the system-wide CPU statistics from all CPUs and combines them into a cgroup_base_stat structure. This is needed because the root cgroup doesn't have a parent to propagate stats from.
755 */
756static void root_cgroup_cputime(struct cgroup_base_stat *bstat)
757{
758 struct task_cputime *cputime = &bstat->cputime;
759 int i;
760
761 memset(bstat, 0, sizeof(*bstat));
762 for_each_possible_cpu(i) {
763 struct kernel_cpustat kcpustat;
764 u64 *cpustat = kcpustat.cpustat;
765 u64 user = 0;
766 u64 sys = 0;
767
768 kcpustat_cpu_fetch(&kcpustat, i);
769
770 user += cpustat[CPUTIME_USER];
771 user += cpustat[CPUTIME_NICE];
772 cputime->utime += user;
773
774 sys += cpustat[CPUTIME_SYSTEM];
775 sys += cpustat[CPUTIME_IRQ];
776 sys += cpustat[CPUTIME_SOFTIRQ];
777 cputime->stime += sys;
778
779 cputime->sum_exec_runtime += user;
780 cputime->sum_exec_runtime += sys;
781
782#ifdef CONFIG_SCHED_CORE
783 bstat->forceidle_sum += cpustat[CPUTIME_FORCEIDLE];
784#endif
785 bstat->ntime += cpustat[CPUTIME_NICE];
786 }
787}
