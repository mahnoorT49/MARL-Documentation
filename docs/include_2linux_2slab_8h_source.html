711static __always_inline enum kmalloc_cache_type kmalloc_type(gfp_t flags, unsigned long caller)
712{
713 /*
714 * The most common case is KMALLOC_NORMAL, so test for it
715 * with a single branch for all the relevant flags.
716 */
717 if (likely((flags & KMALLOC_NOT_NORMAL_BITS) == 0))
718#ifdef CONFIG_RANDOM_KMALLOC_CACHES
719 /* RANDOM_KMALLOC_CACHES_NR (=15) copies + the KMALLOC_NORMAL */
720 return KMALLOC_RANDOM_START + hash_64(caller ^ random_kmalloc_seed,
721 ilog2(RANDOM_KMALLOC_CACHES_NR + 1));
722#else
723 return KMALLOC_NORMAL;
724#endif
725
726 /*
727 * At least one of the flags has to be set. Their priorities in
728 * decreasing order are:
729 * 1) __GFP_DMA
730 * 2) __GFP_RECLAIMABLE
731 * 3) __GFP_ACCOUNT
732 */
733 if (IS_ENABLED(CONFIG_ZONE_DMA) && (flags & __GFP_DMA))
734 return KMALLOC_DMA;
735 if (!IS_ENABLED(CONFIG_MEMCG) || (flags & __GFP_RECLAIMABLE))
736 return KMALLOC_RECLAIM;
737 else
738 return KMALLOC_CGROUP;
739}

757static __always_inline unsigned int __kmalloc_index(size_t size,
758 bool size_is_constant)
759{
760 if (!size)
761 return 0;
762
763 if (size <= KMALLOC_MIN_SIZE)
764 return KMALLOC_SHIFT_LOW;
765
766 if (KMALLOC_MIN_SIZE <= 32 && size > 64 && size <= 96)
767 return 1;
768 if (KMALLOC_MIN_SIZE <= 64 && size > 128 && size <= 192)
769 return 2;
770 if (size <= 8) return 3;
771 if (size <= 16) return 4;
772 if (size <= 32) return 5;
773 if (size <= 64) return 6;
774 if (size <= 128) return 7;
775 if (size <= 256) return 8;
776 if (size <= 512) return 9;
777 if (size <= 1024) return 10;
778 if (size <= 2 * 1024) return 11;
779 if (size <= 4 * 1024) return 12;
780 if (size <= 8 * 1024) return 13;
781 if (size <= 16 * 1024) return 14;
782 if (size <= 32 * 1024) return 15;
783 if (size <= 64 * 1024) return 16;
784 if (size <= 128 * 1024) return 17;
785 if (size <= 256 * 1024) return 18;
786 if (size <= 512 * 1024) return 19;
787 if (size <= 1024 * 1024) return 20;
788 if (size <= 2 * 1024 * 1024) return 21;
789
790 if (!IS_ENABLED(CONFIG_PROFILE_ALL_BRANCHES) && size_is_constant)
791 BUILD_BUG_ON_MSG(1, "unexpected size in kmalloc_index()");
792 else
793 BUG();
794
795 /* Will never be reached. Needed because the compiler may complain */
796 return -1;
797}

1021static __always_inline __alloc_size(1) void *kmalloc_noprof(size_t size, gfp_t flags)
1022{
1023 if (__builtin_constant_p(size) && size) {
1024 unsigned int index;
1025
1026 if (size > KMALLOC_MAX_CACHE_SIZE)
1027 return __kmalloc_large_noprof(size, flags);
1028
1029 index = kmalloc_index(size);
1030 return __kmalloc_cache_noprof(
1031 kmalloc_caches[kmalloc_type(flags, _RET_IP_)][index],
1032 flags, size);
1033 }
1034 return __kmalloc_noprof(size, flags);
1035}
